# >>> MODIFIED[M1]: recursively sanitize BOTH dict keys and string values.
# This fixes dirty config such as "pipeline ", "image_dir ", "text ": "smoke. ".
def sanitize_loaded_config(obj: Any) -> Any:
    if isinstance(obj, dict):
        out: Dict[str, Any] = {}
        for key, value in obj.items():
            clean_key = key.strip() if isinstance(key, str) else str(key).strip()
            out[clean_key] = sanitize_loaded_config(value)
        return out
    if isinstance(obj, list):
        return [sanitize_loaded_config(v) for v in obj]
    if isinstance(obj, str):
        return obj.strip()
    return obj


# >>> MODIFIED[M2]: tolerant text extraction even if spec came from dirty JSON.
def get_prompt_text_from_spec(spec: Dict[str, Any]) -> str:
    if not isinstance(spec, dict):
        return ""
    if "text" in spec and isinstance(spec["text"], str):
        return spec["text"].strip()
    for key, value in spec.items():
        if isinstance(key, str) and key.strip() == "text" and isinstance(value, str):
            return value.strip()
    return ""

def image_to_rgb(image: Image.Image) -> Image.Image:

def list_images(image_dir: Path, recursive: bool) -> List[Path]:
    pats = ("*.jpg", "*.jpeg", "*.png", "*.bmp", "*.webp", "*.tif", "*.tiff")
    items: List[Path] = []
    if recursive:
        for pat in pats:
            items.extend(image_dir.rglob(pat))
    else:
        for pat in pats:
            items.extend(image_dir.glob(pat))
    return sorted({p.resolve() for p in items if p.is_file()})


def clip_box_xyxy(box: Sequence[float], w: int, h: int) -> List[float]:
    x1, y1, x2, y2 = [float(v) for v in box]
    x1 = min(max(x1, 0.0), float(max(w - 1, 0)))
    y1 = min(max(y1, 0.0), float(max(h - 1, 0)))
    x2 = min(max(x2, 0.0), float(max(w - 1, 0)))
    y2 = min(max(y2, 0.0), float(max(h - 1, 0)))
    if x2 < x1:
        x1, x2 = x2, x1
    if y2 < y1:
        y1, y2 = y2, y1
    return [x1, y1, x2, y2]


def box_area(box: Sequence[float]) -> float:
    x1, y1, x2, y2 = [float(v) for v in box]
    return max(0.0, x2 - x1) * max(0.0, y2 - y1)


def intersection_xyxy(a: Sequence[float], b: Sequence[float]) -> float:
    ax1, ay1, ax2, ay2 = [float(v) for v in a]
    bx1, by1, bx2, by2 = [float(v) for v in b]
    ix1 = max(ax1, bx1)
    iy1 = max(ay1, by1)
    ix2 = min(ax2, bx2)
    iy2 = min(ay2, by2)
    return max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)


def compute_iou(a: Sequence[float], b: Sequence[float]) -> float:
    inter = intersection_xyxy(a, b)
    if inter <= 0:
        return 0.0
    ua = box_area(a) + box_area(b) - inter
    return inter / max(ua, 1e-12)


def compute_iof(a: Sequence[float], b: Sequence[float]) -> float:
    inter = intersection_xyxy(a, b)
    if inter <= 0:
        return 0.0
    return inter / max(box_area(a), 1e-12)


def compute_containment_ratio(a: Sequence[float], b: Sequence[float]) -> float:
    inter = intersection_xyxy(a, b)
    return inter / max(min(box_area(a), box_area(b)), 1e-12)


def boxes_are_same_object(
    a: Sequence[float],
    b: Sequence[float],
    iou_thr: float,
    iof_thr: float,
    contain_thr: float,
) -> bool:
    iou = compute_iou(a, b)
    iof_ab = compute_iof(a, b)
    iof_ba = compute_iof(b, a)
    contain = compute_containment_ratio(a, b)
    return (
        iou >= iou_thr
        or iof_ab >= iof_thr
        or iof_ba >= iof_thr
        or contain >= contain_thr
    )


def average_boxes_weighted(dets: List[Detection]) -> List[float]:
    ws = np.array([max(float(d.score), 1e-6) for d in dets], dtype=np.float64)
    boxes = np.array([d.box_xyxy for d in dets], dtype=np.float64)
    avg = np.average(boxes, axis=0, weights=ws)
    return [float(v) for v in avg.tolist()]


def safe_float(v: Any, default: float) -> float:
    try:
        return float(v)
    except Exception:
        return float(default)

def resolve_class_thresholds(cls_cfg: Dict[str, Any], post_cfg: Dict[str, Any]) -> Dict[str, float]:
    return {
        "box_threshold": safe_float(cls_cfg.get("box_threshold", post_cfg.get("box_threshold", 0.25)), 0.25),
        "text_threshold": safe_float(cls_cfg.get("text_threshold", post_cfg.get("text_threshold", 0.18)), 0.18),
        "min_abs_score": safe_float(cls_cfg.get("min_abs_score", post_cfg.get("min_abs_score", 0.0)), 0.0),
        "max_area_ratio": safe_float(cls_cfg.get("max_area_ratio", post_cfg.get("max_area_ratio", 1.0)), 1.0),
        "tau_area_consensus": safe_float(cls_cfg.get("tau_area_consensus", post_cfg.get("tau_area_consensus", 0.0)), 0.0),
    }


# >>> MODIFIED[M3]: prompt normalization uses sanitized text values.
def normalize_prompt_spec(raw_spec: Dict[str, Any], prompt_uid: int) -> Optional[PromptSpec]:
    text = get_prompt_text_from_spec(raw_spec)
    if not text:
        return None
    return PromptSpec(
        text=text,
        family=str(raw_spec.get("family", "full")).strip(),
        hierarchy_level=int(raw_spec.get("hierarchy_level", 0)),
        family_weight=float(raw_spec.get("family_weight", 1.0)),
        region_role=str(raw_spec.get("region_role", "full")).strip(),
        is_primary=bool(raw_spec.get("is_primary", False)),
        prompt_uid=prompt_uid,
    )


def build_runtime_plan(config: Dict[str, Any]) -> RuntimePlan:
    classes_cfg = config.get("classes", {})
    post_cfg = config.get("postprocess", {})
    class_map: Dict[str, Dict[str, Any]] = {}
    all_specs: List[RuntimePrompt] = []
    class_prompt_counts: Dict[str, int] = {}
    next_prompt_uid = 0

    for class_name, class_cfg in classes_cfg.items():
        if not isinstance(class_cfg, dict):
            continue
        class_name = str(class_name).strip()
        class_map[class_name] = class_cfg
        thresholds = resolve_class_thresholds(class_cfg, post_cfg)
        count = 0
        for raw_spec in class_cfg.get("prompt_specs", []) or []:
            prompt_spec = normalize_prompt_spec(raw_spec, prompt_uid=next_prompt_uid)
            if prompt_spec is None:
                LOGGER.warning("Skipping empty/invalid prompt spec for class=%s: %r", class_name, raw_spec)
                continue
            all_specs.append(
                RuntimePrompt(
                    class_name=class_name,
                    class_cfg=class_cfg,
                    thresholds=thresholds,
                    spec=prompt_spec,
                    prompt_uid=next_prompt_uid,
                )
            )
            next_prompt_uid += 1
            count += 1
        class_prompt_counts[class_name] = count

    if not all_specs:
        raise ValueError("No valid prompt_specs found in config['classes']")

    return RuntimePlan(class_map=class_map, all_specs=all_specs, class_prompt_counts=class_prompt_counts)


def load_config(path: Path) -> Dict[str, Any]:
    with path.open("r", encoding="utf-8") as f:
        cfg = json.load(f)
    cfg = sanitize_loaded_config(cfg)  # >>> MODIFIED[M4]
    return cfg

def is_cuda_device(device: torch.device) -> bool:
    return device.type == "cuda"


def configure_runtime_backend(config: Dict[str, Any]) -> None:
    pipe_cfg = config.get("pipeline", {})
    if torch.cuda.is_available():
        if bool(pipe_cfg.get("enable_tf32", False)):
            torch.backends.cuda.matmul.allow_tf32 = True
            torch.backends.cudnn.allow_tf32 = True
        if "cudnn_benchmark" in pipe_cfg:
            torch.backends.cudnn.benchmark = bool(pipe_cfg.get("cudnn_benchmark", False))


def load_model_and_processor(config: Dict[str, Any]) -> Tuple[Any, Any, torch.device]:
    pipe_cfg = config.get("pipeline", {})
    model_dir = pipe_cfg["model_dir"]
    device_name = str(pipe_cfg.get("device", "cuda" if torch.cuda.is_available() else "cpu"))
    local_files_only = bool(pipe_cfg.get("local_files_only", True))
    torch_dtype_name = str(pipe_cfg.get("torch_dtype", "float16")).lower()

    device = torch.device(device_name)

    processor = AutoProcessor.from_pretrained(
        model_dir,
        local_files_only=local_files_only
    )

    # 关键修复：
    # 不再把整个模型直接加载为 fp16/bf16，统一先用 fp32 权重，避免内部 Float/Half 冲突
    model = AutoModelForZeroShotObjectDetection.from_pretrained(
        model_dir,
        local_files_only=local_files_only,
        torch_dtype=torch.float32,
        low_cpu_mem_usage=True,
    )

    model.to(device)
    model.eval()

    # 用模型属性保存 AMP dtype，后面 forward 时读取
    amp_dtype = None
    if device.type == "cuda":
        if torch_dtype_name in ("bfloat16", "bf16"):
            if torch.cuda.is_bf16_supported():
                amp_dtype = torch.bfloat16
            else:
                LOGGER.warning("bf16 requested but not supported on this GPU, fallback to fp32.")
        elif torch_dtype_name in ("float16", "fp16"):
            amp_dtype = torch.float16
        elif torch_dtype_name in ("float32", "fp32"):
            amp_dtype = None

    model._amp_dtype = amp_dtype

    if bool(pipe_cfg.get("torch_compile", False)) and hasattr(torch, "compile"):
        mode = str(pipe_cfg.get("torch_compile_mode", "reduce-overhead"))
        LOGGER.info("Compiling model with torch.compile(mode=%s)", mode)
        model = torch.compile(model, mode=mode)

    return processor, model, device

def _slice_grounding_dino_outputs(outputs: Any, index: int) -> SimpleNamespace:
    return SimpleNamespace(
        logits=outputs.logits[index:index + 1],
        pred_boxes=outputs.pred_boxes[index:index + 1],
    )


# >>> MODIFIED[M5]: keep matching input_ids for each prompt during batched post-process.
def _slice_input_ids(input_ids: torch.Tensor, index: int) -> torch.Tensor:
    return input_ids[index:index + 1]


def _maybe_channels_last(inputs: Dict[str, torch.Tensor], enabled: bool) -> Dict[str, torch.Tensor]:
    if not enabled:
        return inputs
    out = dict(inputs)
    if "pixel_values" in out and torch.is_floating_point(out["pixel_values"]):
        out["pixel_values"] = out["pixel_values"].contiguous(memory_format=torch.channels_last)
    return out

def _move_inputs_to_device(inputs: Dict[str, torch.Tensor], device: torch.device) -> Dict[str, torch.Tensor]:
    return {k: v.to(device=device) for k, v in inputs.items()}

def _autocast_context(model, device: torch.device):
    amp_dtype = getattr(model, "_amp_dtype", None)
    if device.type == "cuda" and amp_dtype is not None:
        return torch.autocast(device_type="cuda", dtype=amp_dtype)
    return nullcontext()

@torch.inference_mode()
def run_grounding_dino(
    image: Image.Image,
    prompt: str,
    processor,
    model,
    device: torch.device,
    box_threshold: float,
    text_threshold: float,
    channels_last: bool = True,
) -> Dict[str, Any]:
    image = image_to_rgb(image)
    w, h = image.size
    inputs = processor(images=image, text=prompt, return_tensors="pt")

    # 不手动转 half，只搬到 device
    inputs = _move_inputs_to_device(inputs, device)
    inputs = _maybe_channels_last(inputs, channels_last)

    with _autocast_context(model, device):
        outputs = model(**inputs)

    post = processor.post_process_grounded_object_detection(
        outputs,
        inputs["input_ids"],
        box_threshold=float(box_threshold),
        text_threshold=float(text_threshold),
        target_sizes=[(h, w)],
    )
    return post[0]


@torch.inference_mode()
def run_grounding_dino_batch(
    image: Image.Image,
    prompts: List[str],
    processor,
    model,
    device: torch.device,
    box_threshold: float,
    text_threshold: float,
    channels_last: bool = True,
    per_prompt_thresholds: Optional[List[Tuple[float, float]]] = None,
) -> List[Dict[str, Any]]:
    """
    Same-view multi-prompt batch inference with binary fallback on failure.
    """
    if not prompts:
        return []
    if per_prompt_thresholds is not None and len(per_prompt_thresholds) != len(prompts):
        raise ValueError("per_prompt_thresholds length must match prompts length")

    if len(prompts) == 1:
        if per_prompt_thresholds is None:
            bthr, tthr = float(box_threshold), float(text_threshold)
        else:
            bthr, tthr = per_prompt_thresholds[0]
        return [
            run_grounding_dino(
                image=image,
                prompt=prompts[0],
                processor=processor,
                model=model,
                device=device,
                box_threshold=bthr,
                text_threshold=tthr,
                channels_last=channels_last,
            )
        ]

    image = image_to_rgb(image)
    w, h = image.size
    images = [image for _ in prompts]

    try:
        inputs = processor(images=images, text=prompts, return_tensors="pt", padding=True)

        # 不手动转 half，只搬到 device
        inputs = _move_inputs_to_device(inputs, device)
        inputs = _maybe_channels_last(inputs, channels_last)

        with _autocast_context(model, device):
            outputs = model(**inputs)

        results: List[Dict[str, Any]] = []
        for idx in range(len(prompts)):
            if per_prompt_thresholds is None:
                bthr = float(box_threshold)
                tthr = float(text_threshold)
            else:
                bthr, tthr = map(float, per_prompt_thresholds[idx])

            out_i = _slice_grounding_dino_outputs(outputs, idx)
            input_ids_i = _slice_input_ids(inputs["input_ids"], idx)

            post = processor.post_process_grounded_object_detection(
                out_i,
                input_ids_i,
                box_threshold=bthr,
                text_threshold=tthr,
                target_sizes=[(h, w)],
            )
            results.append(post[0])
        return results

    except RuntimeError as exc:
        err = str(exc).lower()
        if "out of memory" not in err and "cuda" not in err:
            raise
        LOGGER.warning("Batch inference failed, fallback split=%d: %s", len(prompts), exc)
        if device.type == "cuda":
            torch.cuda.empty_cache()
        if len(prompts) == 1:
            raise
        mid = len(prompts) // 2
        left = run_grounding_dino_batch(
            image=image,
            prompts=prompts[:mid],
            processor=processor,
            model=model,
            device=device,
            box_threshold=box_threshold,
            text_threshold=text_threshold,
            channels_last=channels_last,
            per_prompt_thresholds=None if per_prompt_thresholds is None else per_prompt_thresholds[:mid],
        )
        right = run_grounding_dino_batch(
            image=image,
            prompts=prompts[mid:],
            processor=processor,
            model=model,
            device=device,
            box_threshold=box_threshold,
            text_threshold=text_threshold,
            channels_last=channels_last,
            per_prompt_thresholds=None if per_prompt_thresholds is None else per_prompt_thresholds[mid:],
        )
        return left + right

def _resize_image_keep_rgb(image: Image.Image, scale: float) -> Image.Image:
    if abs(scale - 1.0) < 1e-8:
        return image
    w, h = image.size
    nw = max(1, int(round(w * scale)))
    nh = max(1, int(round(h * scale)))
    return image.resize((nw, nh), resample=Image.BILINEAR)


def make_tta_views(image: Image.Image, post_cfg: Dict[str, Any]) -> List[TTAView]:
    tta_cfg = post_cfg.get("tta", {}) or {}
    enabled = bool(tta_cfg.get("enabled", False))
    if not enabled:
        return [TTAView(name="orig", image=image, inverse_map=lambda b: [float(v) for v in b])]

    hflip = bool(tta_cfg.get("hflip", False))
    scales = [float(s) for s in (tta_cfg.get("scales", []) or [])]
    scales = [s for s in scales if s > 0]
    if 1.0 not in scales:
        scales = [1.0] + scales

    orig_w, orig_h = image.size
    views: List[TTAView] = []

    for scale in scales:
        scaled = _resize_image_keep_rgb(image, scale)

        def _make_scale_inverse(s: float) -> Callable[[Sequence[float]], List[float]]:
            def _inv(box: Sequence[float]) -> List[float]:
                x1, y1, x2, y2 = [float(v) / s for v in box]
                return clip_box_xyxy([x1, y1, x2, y2], orig_w, orig_h)
            return _inv

        views.append(TTAView(name=f"scale_{scale:g}", image=scaled, inverse_map=_make_scale_inverse(scale)))

        if hflip:
            flipped = scaled.transpose(Image.FLIP_LEFT_RIGHT)

            def _make_flip_scale_inverse(s: float, flip_w: int) -> Callable[[Sequence[float]], List[float]]:
                def _inv(box: Sequence[float]) -> List[float]:
                    x1, y1, x2, y2 = [float(v) for v in box]
                    fx1 = float(flip_w) - x2
                    fx2 = float(flip_w) - x1
                    fx1 /= s
                    fx2 /= s
                    y1 /= s
                    y2 /= s
                    return clip_box_xyxy([fx1, y1, fx2, y2], orig_w, orig_h)
                return _inv

            views.append(
                TTAView(
                    name=f"scale_{scale:g}_hflip",
                    image=flipped,
                    inverse_map=_make_flip_scale_inverse(scale, scaled.size[0]),
                )
            )

    return views

def _to_list(obj: Any) -> List[Any]:
    if obj is None:
        return []
    if isinstance(obj, torch.Tensor):
        return obj.detach().cpu().tolist()
    if isinstance(obj, np.ndarray):
        return obj.tolist()
    if isinstance(obj, (list, tuple)):
        return list(obj)
    return [obj]


def _extract_detection_fields(result: Dict[str, Any]) -> Tuple[List[List[float]], List[float], List[Any], List[Any]]:
    boxes = _to_list(result.get("boxes", []))
    scores = _to_list(result.get("scores", []))
    labels = _to_list(result.get("labels", []))
    text_labels = _to_list(result.get("text_labels", result.get("phrases", [])))
    if not text_labels and len(labels) > 0:
        text_labels = [None] * len(labels)
    return boxes, scores, labels, text_labels


# >>> MODIFIED[M7]: prompt_hits are tracked once per ORIGINAL image; not once per TTA view.
# >>> MODIFIED[M8]: per-class score/area filters are enforced after batched forward.
def collect_all_class_detections(
    image: Image.Image,
    runtime_plan: RuntimePlan,
    processor,
    model,
    device: torch.device,
    pipeline_cfg: Dict[str, Any],
    post_cfg: Dict[str, Any],
    tta_views: List[TTAView],
) -> Tuple[Dict[str, List[Detection]], Dict[str, int], int]:
    w, h = image.size
    prompt_batch_size = max(1, int(post_cfg.get("prompt_batch_size", 8)))
    channels_last = bool(pipeline_cfg.get("channels_last", True))

    class_dets: Dict[str, List[Detection]] = {cn: [] for cn in runtime_plan.class_map.keys()}
    raw_detection_count = 0

    hit_prompt_uids: Dict[str, Set[int]] = {cn: set() for cn in runtime_plan.class_map.keys()}

    for view in tta_views:
        for batch_start in range(0, len(runtime_plan.all_specs), prompt_batch_size):
            batch = runtime_plan.all_specs[batch_start:batch_start + prompt_batch_size]
            batch_prompts = [rp.spec.text for rp in batch]
            batch_thresholds = [
                (float(rp.thresholds["box_threshold"]), float(rp.thresholds["text_threshold"]))
                for rp in batch
            ]

            default_box_thr = min(x[0] for x in batch_thresholds)
            default_text_thr = min(x[1] for x in batch_thresholds)

            batch_outputs = run_grounding_dino_batch(
                image=view.image,
                prompts=batch_prompts,
                processor=processor,
                model=model,
                device=device,
                box_threshold=default_box_thr,
                text_threshold=default_text_thr,
                channels_last=channels_last,
                per_prompt_thresholds=batch_thresholds,
            )

            for rp, result in zip(batch, batch_outputs):
                thresholds = rp.thresholds
                boxes, scores, labels, text_labels = _extract_detection_fields(result)
                kept_this_prompt = False

                for i, (box, score) in enumerate(zip(boxes, scores)):
                    score_f = float(score)
                    if score_f < float(thresholds["min_abs_score"]):
                        continue

                    box_xyxy_view = [float(v) for v in box]
                    box_xyxy_orig = view.inverse_map(box_xyxy_view)
                    box_xyxy_orig = clip_box_xyxy(box_xyxy_orig, w, h)

                    area_ratio = box_area(box_xyxy_orig) / max(float(w * h), 1.0)
                    if area_ratio > float(thresholds["max_area_ratio"]):
                        continue

                    det = Detection(
                        class_name=rp.class_name,
                        prompt_text=rp.spec.text,
                        prompt_uid=rp.prompt_uid,
                        score=score_f,
                        box_xyxy=box_xyxy_orig,
                        family=rp.spec.family,
                        hierarchy_level=rp.spec.hierarchy_level,
                        family_weight=rp.spec.family_weight,
                        region_role=rp.spec.region_role,
                        is_primary=rp.spec.is_primary,
                        source_view=view.name,
                        raw_label=labels[i] if i < len(labels) else None,
                        matched_text=text_labels[i] if i < len(text_labels) else None,
                        extra={
                            "box_threshold": float(thresholds["box_threshold"]),
                            "text_threshold": float(thresholds["text_threshold"]),
                            "min_abs_score": float(thresholds["min_abs_score"]),
                            "max_area_ratio": float(thresholds["max_area_ratio"]),
                            "area_ratio": float(area_ratio),
                        },
                    )
                    class_dets[rp.class_name].append(det)
                    raw_detection_count += 1
                    kept_this_prompt = True

                if kept_this_prompt:
                    hit_prompt_uids[rp.class_name].add(rp.prompt_uid)

    prompt_hits = {cn: len(hit_prompt_uids[cn]) for cn in runtime_plan.class_map.keys()}
    return class_dets, prompt_hits, raw_detection_count

def dedup_detections_for_class(class_name: str, dets: List[Detection], post_cfg: Dict[str, Any]) -> List[Detection]:
    if not dets:
        return []

    dedup_iou = safe_float(post_cfg.get("dedup_iou", 0.55), 0.55)
    dedup_iof = safe_float(post_cfg.get("dedup_iof", 0.85), 0.85)
    dedup_contain = safe_float(post_cfg.get("containment_ratio", 0.90), 0.90)
    canonical_mode = str(post_cfg.get("canonical_box_mode", "best")).strip().lower()

    ordered = sorted(
        dets,
        key=lambda d: (
            -float(d.score),
            -float(d.family_weight),
            -int(bool(d.is_primary)),
            -int(d.hierarchy_level),
        ),
    )

    groups: List[List[Detection]] = []
    for det in ordered:
        placed = False
        for group in groups:
            ref = group[0]
            if boxes_are_same_object(ref.box_xyxy, det.box_xyxy, dedup_iou, dedup_iof, dedup_contain):
                group.append(det)
                placed = True
                break
        if not placed:
            groups.append([det])

    merged: List[Detection] = []
    for group in groups:
        best = max(group, key=lambda d: (float(d.score), float(d.family_weight), int(d.is_primary), d.hierarchy_level))
        merged_box = best.box_xyxy if canonical_mode == "best" else average_boxes_weighted(group)
        merged_det = Detection(
            class_name=class_name,
            prompt_text=best.prompt_text,
            prompt_uid=best.prompt_uid,
            score=float(max(d.score for d in group)),
            box_xyxy=[float(v) for v in merged_box],
            family=best.family,
            hierarchy_level=best.hierarchy_level,
            family_weight=best.family_weight,
            region_role=best.region_role,
            is_primary=best.is_primary,
            source_view=best.source_view,
            raw_label=best.raw_label,
            matched_text=best.matched_text,
            extra={
                "group_size": len(group),
                "source_views": sorted({d.source_view for d in group}),
                "support_scores": [float(d.score) for d in group],
                "support_prompts": [d.prompt_text for d in group],
            },
        )
        merged.append(merged_det)

    max_final = int(post_cfg.get("max_final_boxes_per_class", 0) or 0)
    merged = sorted(merged, key=lambda d: -float(d.score))
    if max_final > 0:
        merged = merged[:max_final]
    return merged


def finalize_all_classes(class_dets: Dict[str, List[Detection]], post_cfg: Dict[str, Any]) -> Dict[str, List[Detection]]:
    return {class_name: dedup_detections_for_class(class_name, dets, post_cfg) for class_name, dets in class_dets.items()}

def draw_detections(image: Image.Image, detections: Dict[str, List[Detection]]) -> Image.Image:
    image = image.copy().convert("RGB")
    draw = ImageDraw.Draw(image)
    font = _load_default_font()
    for class_name, dets in detections.items():
        for det in dets:
            x1, y1, x2, y2 = [float(v) for v in det.box_xyxy]

"""
把 GDINO 的 final_dets 顺手导出为课程学习脚本可读的 sidecar JSON。

用法：
1) 把本文件复制到你的项目里；
2) 在 GDINO 推理循环里拿到 final_dets 后调用 export_sidecar(...)

不会增加额外推理，仅仅是把你当前内存中的结果序列化到磁盘。

from __future__ import annotations

import json
from pathlib import Path
from typing import Any, Dict, Iterable, List, Optional


def _obj_to_dict(obj: Any) -> Dict[str, Any]:
    if isinstance(obj, dict):
        return obj
    out: Dict[str, Any] = {}
    for key in dir(obj):
        if key.startswith("_"):
            continue
        try:
            value = getattr(obj, key)
        except Exception:
            continue
        if callable(value):
            continue
        out[key] = value
    return out


def _safe_list4(v: Any) -> Optional[List[float]]:
    if not isinstance(v, (list, tuple)) or len(v) != 4:
        return None
    try:
        return [float(x) for x in v]
    except Exception:
        return None


def export_sidecar(
    final_dets: Iterable[Any],
    image_path: str,
    image_root: str,
    output_dir: str,
    image_width: int,
    image_height: int,
    extra: Optional[Dict[str, Any]] = None,
) -> Path:
    image_path_p = Path(image_path)
    image_root_p = Path(image_root)
    rel = image_path_p.relative_to(image_root_p).with_suffix(".json")
    sidecar_path = Path(output_dir) / "labels_meta_raw" / rel
    sidecar_path.parent.mkdir(parents=True, exist_ok=True)

    detections: List[Dict[str, Any]] = []
    for det in final_dets:
        d = _obj_to_dict(det)
        box_xyxy = (
            _safe_list4(d.get("box_xyxy"))
            or _safe_list4(d.get("meta_fused_box_xyxy"))
            or _safe_list4(d.get("primary_box_xyxy"))
        )
        if box_xyxy is None:
            continue

        qbd = d.get("quality_breakdown") if isinstance(d.get("quality_breakdown"), dict) else {}
        item = {
            "class_id": d.get("class_id"),
            "class_name": d.get("class_name"),
            "box_xyxy": box_xyxy,
            "score": d.get("score", d.get("meta_fused_score", d.get("primary_score", 0.0))),
            "quality_q": d.get("quality_q", 0.0),
            "support_prompt_count": d.get("support_prompt_count", 0),
            "support_prompt_ratio": d.get("support_prompt_ratio", 0.0),
            "prompt_hit_count_in_image": d.get("prompt_hit_count_in_image", 0),
            "teacher_risk_hint": d.get("teacher_risk_hint", "unknown"),
            "selected_level": d.get("selected_level", d.get("region_role", d.get("family", "unknown"))),
            "area_ratio": d.get("area_ratio", 0.0),
            "box_dispersion": d.get("box_dispersion", 0.0),
            "local_evidence_q": d.get("local_evidence_q", qbd.get("s_sem", 0.0)),
            "p_drift": d.get("p_drift", qbd.get("p_drift", 0.0)),

        }
        detections.append(item)

    payload = {
        "image_path": str(image_path_p),
        "image_width": int(image_width),
        "image_height": int(image_height),
        "detections": detections,
    }

        payload.update(extra)

    with sidecar_path.open("w", encoding="utf-8") as f:
        json.dump(payload, f, ensure_ascii=False, indent=2)
    return sidecar_path

def sanitize_keys(obj: Any) -> Any:
    if isinstance(obj, dict):
        out: Dict[str, Any] = {}
        for k, v in obj.items():
            kk = k.strip() if isinstance(k, str) else k
            out[kk] = sanitize_keys(v)
        return out
    if isinstance(obj, list):
        return [sanitize_keys(i) for i in obj]
    if isinstance(obj, str):
        return obj.strip()
    return obj


def load_json(path: Path) -> Any:
    with path.open("r", encoding="utf-8") as f:
        data = json.load(f)
    return sanitize_keys(data)


def dump_json(path: Path, obj: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        json.dump(obj, f, ensure_ascii=False, indent=2)


def dump_text(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding="utf-8")


def dump_yaml(path: Path, data: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        yaml.safe_dump(data, f, allow_unicode=True, sort_keys=False)


def clamp01(x: float) -> float:
    if math.isnan(x):
        return 0.0
    return max(0.0, min(1.0, x))


def safe_float(v: Any, default: float = 0.0) -> float:
    try:
        return float(v)
    except Exception:
        return default


def safe_int(v: Any, default: int = 0) -> int:
    try:
        return int(v)
    except Exception:
        return default


def ensure_parent(path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)


def maybe_symlink_or_copy(src: Path, dst: Path, mode: str = "symlink") -> None:
    ensure_parent(dst)
    if dst.exists() or dst.is_symlink():
        if dst.is_symlink() or dst.is_file():
            dst.unlink()
        elif dst.is_dir():
            shutil.rmtree(dst)
    if mode == "copy":
        shutil.copy2(src, dst)
        return
    try:
        os.symlink(src.resolve(), dst)
    except Exception:
        shutil.copy2(src, dst)


def read_image_size(image_path: Path) -> Tuple[int, int]:
    with Image.open(image_path) as im:
        return im.size


def quantile(values: Sequence[float], q: float) -> float:
    if not values:
        return 0.0
    xs = sorted(values)
    if q <= 0:
        return xs[0]
    if q >= 1:
        return xs[-1]
    pos = (len(xs) - 1) * q
    lo = int(math.floor(pos))
    hi = int(math.ceil(pos))
    if lo == hi:
        return xs[lo]
    w = pos - lo
    return xs[lo] * (1.0 - w) + xs[hi] * w

def discover_images(image_dir: Path, recursive: bool = True) -> Dict[str, Path]:

def load_names_map(cfg: Dict[str, Any]) -> Tuple[Dict[str, int], Dict[int, str]]:
    classes = cfg.get("classes", {})
    if not classes:
        raise ValueError("config 中必须提供 classes，例如 {'fire': 0, 'smoke': 1}")
    name_to_id: Dict[str, int] = {}
    id_to_name: Dict[int, str] = {}
    for name, cid in classes.items():
        cid_int = safe_int(cid)
        name_to_id[str(name)] = cid_int
        id_to_name[cid_int] = str(name)
    return name_to_id, id_to_name

def pick_detections_container(meta: Any) -> List[Dict[str, Any]]:
    if isinstance(meta, list):
        return [x for x in meta if isinstance(x, dict)]
    if not isinstance(meta, dict):
        return []

    if "detections" in meta and isinstance(meta["detections"], dict):
        result: List[Dict[str, Any]] = []
        for dets_list in meta["detections"].values():
            if isinstance(dets_list, list):
                result.extend([x for x in dets_list if isinstance(x, dict)])
        return result

    for key in ("detections", "final_detections", "final_dets", "labels", "annotations", "objects"):
        value = meta.get(key)
        if isinstance(value, list):
            return [x for x in value if isinstance(x, dict)]
    return []


def normalize_xywhn(xywhn: Sequence[float]) -> List[float]:
    if len(xywhn) != 4:
        raise ValueError("xywhn 必须是长度为 4 的数组")
    x, y, w, h = [safe_float(v) for v in xywhn]
    return [clamp01(x), clamp01(y), clamp01(w), clamp01(h)]


def xyxy_to_xywhn(box_xyxy: Sequence[float], width: int, height: int) -> List[float]:
    if len(box_xyxy) != 4:
        raise ValueError("box_xyxy 必须是长度为 4 的数组")
    x1, y1, x2, y2 = [safe_float(v) for v in box_xyxy]
    x1 = max(0.0, min(x1, width))
    x2 = max(0.0, min(x2, width))
    y1 = max(0.0, min(y1, height))
    y2 = max(0.0, min(y2, height))
    if x2 < x1:
        x1, x2 = x2, x1
    if y2 < y1:
        y1, y2 = y2, y1
    bw = max(0.0, x2 - x1)
    bh = max(0.0, y2 - y1)
    cx = x1 + bw / 2.0
    cy = y1 + bh / 2.0
    return [
        clamp01(cx / max(1.0, width)),
        clamp01(cy / max(1.0, height)),
        clamp01(bw / max(1.0, width)),
        clamp01(bh / max(1.0, height)),
    ]


def xywhn_to_xyxy(xywhn: Sequence[float], width: int, height: int) -> List[float]:
    if len(xywhn) != 4:
        raise ValueError("xywhn 必须是长度为 4 的数组")
    cx, cy, bw, bh = [safe_float(v) for v in xywhn]
    cx *= width
    cy *= height
    bw *= width
    bh *= height
    x1 = max(0.0, cx - bw / 2.0)
    y1 = max(0.0, cy - bh / 2.0)
    x2 = min(float(width), cx + bw / 2.0)
    y2 = min(float(height), cy + bh / 2.0)
    return [x1, y1, x2, y2]


def extract_box_and_normalize(det: Dict[str, Any], width: int, height: int) -> Tuple[List[float], List[float]]:
    box_xyxy: Optional[List[float]] = None
    xywhn: Optional[List[float]] = None

    if isinstance(det.get("xywhn"), (list, tuple)):
        xywhn = normalize_xywhn(det["xywhn"])
        box_xyxy = xywhn_to_xyxy(xywhn, width, height)
    elif isinstance(det.get("bbox_xywhn"), (list, tuple)):
        xywhn = normalize_xywhn(det["bbox_xywhn"])
        box_xyxy = xywhn_to_xyxy(xywhn, width, height)
    else:
        for key in ("box_xyxy", "xyxy", "bbox_xyxy", "bbox"):
            value = det.get(key)
            if isinstance(value, (list, tuple)) and len(value) == 4:
                box_xyxy = [safe_float(v) for v in value]
                xywhn = xyxy_to_xywhn(box_xyxy, width, height)
                break

    if xywhn is None or box_xyxy is None:
        raise ValueError("检测框里找不到 box_xyxy / xyxy / xywhn")

    return box_xyxy, xywhn


def infer_prompt_denominator(cfg: Dict[str, Any], class_name: str, raw: Dict[str, Any]) -> int:
    pseudo_cfg = cfg.get("pseudo_label", {}) or {}
    per_class_prompt_cnt = (pseudo_cfg.get("prompt_count_by_class", {}) or {}).get(class_name)
    if per_class_prompt_cnt is not None:
        return max(1, safe_int(per_class_prompt_cnt, 1))

    explicit_total = raw.get("prompt_count_total")
    if explicit_total is not None:
        return max(1, safe_int(explicit_total, 1))

    prompt_lists = [
        raw.get("all_prompts"),
        raw.get("candidate_prompts"),
        raw.get("kept_candidate_prompts"),
    ]
    for item in prompt_lists:
        if isinstance(item, list) and item:
            return max(1, len(item))

    default_cnt = pseudo_cfg.get("default_prompt_count", 4)
    return max(1, safe_int(default_cnt, 4))


def normalize_detection_fields(raw_in: Dict[str, Any], cfg: Dict[str, Any], class_name: str) -> Dict[str, Any]:
    raw = dict(raw_in)
    extra = raw.get("extra", {}) if isinstance(raw.get("extra"), dict) else {}
    qbd = raw.get("quality_breakdown") if isinstance(raw.get("quality_breakdown"), dict) else {}

    if raw.get("area_ratio") is None:
        raw["area_ratio"] = safe_float(extra.get("area_ratio"), 0.0)

    support_prompts = raw.get("support_prompts")
    if not isinstance(support_prompts, list):
        support_prompts = extra.get("support_prompts") if isinstance(extra.get("support_prompts"), list) else []
    support_count_guess = len(support_prompts)

    if raw.get("support_prompt_count") is None:
        raw["support_prompt_count"] = support_count_guess
    raw["support_prompt_count"] = max(0, safe_int(raw.get("support_prompt_count"), support_count_guess))

    prompt_hit_default = raw["support_prompt_count"]
    if raw.get("prompt_hit_count_in_image") is None:
        raw["prompt_hit_count_in_image"] = prompt_hit_default
    raw["prompt_hit_count_in_image"] = max(0, safe_int(raw.get("prompt_hit_count_in_image"), prompt_hit_default))

    if raw.get("teacher_risk_hint") is None:
        family = str(raw.get("family", raw.get("region_role", "unknown"))).strip().lower()
        is_primary = bool(raw.get("is_primary", False))
        if is_primary:
            raw["teacher_risk_hint"] = "low"
        elif family in {"full", "body", "attribute"}:
            raw["teacher_risk_hint"] = "medium"
        else:
            raw["teacher_risk_hint"] = "unknown"
    raw["teacher_risk_hint"] = str(raw.get("teacher_risk_hint", "unknown")).strip().lower()

    if raw.get("selected_level") is None:
        raw["selected_level"] = raw.get("region_role", raw.get("family", "unknown"))
    raw["selected_level"] = str(raw.get("selected_level", "unknown")).strip().lower()

    if raw.get("score") is None:
        raw["score"] = raw.get("meta_fused_score", raw.get("primary_score", 0.0))
    raw["score"] = clamp01(safe_float(raw.get("score"), 0.0))

    if raw.get("quality_q") is None:
        raw["quality_q"] = raw.get("q", raw["score"])
    raw["quality_q"] = clamp01(safe_float(raw.get("quality_q"), raw["score"]))

    if raw.get("support_prompt_ratio") is None:
        prompt_den = infer_prompt_denominator(cfg, class_name, raw)
        raw["support_prompt_ratio"] = raw["support_prompt_count"] / max(1.0, float(prompt_den))
    raw["support_prompt_ratio"] = clamp01(safe_float(raw.get("support_prompt_ratio"), 0.0))

    if raw.get("local_evidence_q") is None:
        raw["local_evidence_q"] = qbd.get("s_sem", 0.0)
    raw["local_evidence_q"] = clamp01(safe_float(raw.get("local_evidence_q"), 0.0))

    if raw.get("p_drift") is None:
        raw["p_drift"] = qbd.get("p_drift", 0.0)
    raw["p_drift"] = clamp01(safe_float(raw.get("p_drift"), 0.0))

    raw["box_dispersion"] = clamp01(safe_float(raw.get("box_dispersion"), 0.0))
    raw["area_ratio"] = clamp01(safe_float(raw.get("area_ratio"), 0.0))
    return raw


def compute_course_q(
    det: Dict[str, Any],
    weights: Dict[str, float],
    total_views: int,
) -> Tuple[float, Dict[str, float]]:
    quality_q = clamp01(safe_float(det.get("quality_q"), safe_float(det.get("q"), 0.0)))
    score = clamp01(safe_float(det.get("score"), safe_float(det.get("primary_score"), quality_q)))
    support_prompt_count = max(0, safe_int(det.get("support_prompt_count"), 0))
    support_prompt_ratio = clamp01(safe_float(det.get("support_prompt_ratio"), 0.0))
    prompt_hit_count = max(0, safe_int(det.get("prompt_hit_count_in_image"), support_prompt_count))
    box_dispersion = clamp01(safe_float(det.get("box_dispersion"), 0.0))
    area_ratio = clamp01(safe_float(det.get("area_ratio"), 0.0))

    qb = det.get("quality_breakdown") if isinstance(det.get("quality_breakdown"), dict) else {}
    local_evidence_q = clamp01(safe_float(det.get("local_evidence_q"), safe_float(qb.get("s_sem"), 0.0)))
    drift = clamp01(safe_float(det.get("p_drift"), safe_float(qb.get("p_drift"), 0.0)))
    teacher_risk_hint = str(det.get("teacher_risk_hint", det.get("risk_hint", "unknown"))).strip().lower()
    selected_level = str(det.get("selected_level", det.get("region_role", det.get("family", "unknown")))).strip().lower()

    tta_hits = max(0, safe_int(det.get("tta_hits"), 0))
    if tta_hits <= 0 and total_views > 1:
        tta_hits = 1 if prompt_hit_count > 0 else 0
    tta_ratio = clamp01(tta_hits / max(1.0, float(total_views)))

    prompt_support = max(
        support_prompt_ratio,
        clamp01(support_prompt_count / 4.0),
        clamp01(prompt_hit_count / 4.0),
    )
    stab = 1.0 - box_dispersion
    drift_ok = 1.0 - drift
    area_ok = 1.0 - max(0.0, area_ratio - 0.5) * 2.0
    area_ok = clamp01(area_ok)
    risk_penalty = RISK_PENALTY.get(teacher_risk_hint, 0.05)
    level_bonus = LEVEL_BONUS.get(selected_level, 0.0)

    components = {
        "quality_q": quality_q,
        "score": score,
        "prompt_support": prompt_support,
        "tta_ratio": tta_ratio,
        "local_evidence_q": local_evidence_q,
        "stability": stab,
        "drift_ok": drift_ok,
        "area_ok": area_ok,
        "risk_penalty": risk_penalty,
        "level_bonus": level_bonus,
    }

    total = 0.0
    total += weights.get("quality_q", 0.55) * quality_q
    total += weights.get("score", 0.20) * score
    total += weights.get("prompt_support", 0.10) * prompt_support
    total += weights.get("tta_ratio", 0.05) * tta_ratio
    total += weights.get("local_evidence_q", 0.04) * local_evidence_q
    total += weights.get("stability", 0.04) * stab
    total += weights.get("drift_ok", 0.01) * drift_ok
    total += weights.get("area_ok", 0.01) * area_ok
    total += weights.get("level_bonus", 1.0) * level_bonus
    total -= weights.get("risk_penalty", 1.0) * risk_penalty

    return clamp01(total), components


def parse_one_meta(
    meta_path: Path,
    image_path: Path,
    rel_stem: str,
    cfg: Dict[str, Any],
    name_to_id: Dict[str, int],
) -> Optional[ImageRecord]:
    meta = load_json(meta_path)

    image_w = safe_int(meta.get("image_width"), 0) if isinstance(meta, dict) else 0
    image_h = safe_int(meta.get("image_height"), 0) if isinstance(meta, dict) else 0
    if image_w <= 0 or image_h <= 0:
        img_size = meta.get("image_size", {}) if isinstance(meta, dict) else {}
        if isinstance(img_size, dict):
            image_w = safe_int(img_size.get("width"), 0)
            image_h = safe_int(img_size.get("height"), 0)
    if image_w <= 0 or image_h <= 0:
        image_w, image_h = read_image_size(image_path)

    total_views = safe_int((cfg.get("pseudo_label") or {}).get("tta_total_views"), 1)
    if total_views <= 0:
        total_views = 1

    det_dicts = pick_detections_container(meta)
    q_weights = (cfg.get("pseudo_label") or {}).get("q_weights", {}) or {}
    global_cfg = cfg.get("pseudo_label", {}) or {}

    dets: List[CurriculumDet] = []
    for raw0 in det_dicts:
        class_name = raw0.get("class_name")
        class_id = raw0.get("class_id")
        if class_name is None and class_id is None:
            continue
        if class_name is None:
            class_name = next((n for n, cid in name_to_id.items() if cid == safe_int(class_id)), None)
        if class_name is None or str(class_name) not in name_to_id:
            continue
        class_name = str(class_name)
        class_id = name_to_id[class_name]

        raw = normalize_detection_fields(raw0, cfg, class_name)
        try:
            box_xyxy, xywhn = extract_box_and_normalize(raw, image_w, image_h)
        except Exception:
            continue

        course_q, _components = compute_course_q(raw, q_weights, total_views)
        score = clamp01(safe_float(raw.get("score"), safe_float(raw.get("primary_score"), 0.0)))
        quality_q = clamp01(safe_float(raw.get("quality_q"), safe_float(raw.get("q"), score)))
        support_prompt_count = max(0, safe_int(raw.get("support_prompt_count"), 0))
        support_prompt_ratio = clamp01(safe_float(raw.get("support_prompt_ratio"), 0.0))
        prompt_hit_count = max(0, safe_int(raw.get("prompt_hit_count_in_image"), support_prompt_count))
        box_dispersion = clamp01(safe_float(raw.get("box_dispersion"), 0.0))
        teacher_risk_hint = str(raw.get("teacher_risk_hint", raw.get("risk_hint", "unknown"))).strip().lower()
        selected_level = str(raw.get("selected_level", raw.get("region_role", raw.get("family", "unknown")))).strip().lower()
        area_ratio = clamp01(safe_float(raw.get("area_ratio"), 0.0))
        qb = raw.get("quality_breakdown") if isinstance(raw.get("quality_breakdown"), dict) else {}
        local_evidence_q = clamp01(safe_float(raw.get("local_evidence_q"), safe_float(qb.get("s_sem"), 0.0)))
        drift = clamp01(safe_float(raw.get("p_drift"), safe_float(qb.get("p_drift"), 0.0)))

        class_cfg = get_class_cfg(cfg, class_name)
        min_score = safe_float(class_cfg.get("min_score"), safe_float(global_cfg.get("min_score"), 0.0))
        min_quality_q = safe_float(class_cfg.get("min_quality_q"), safe_float(global_cfg.get("min_quality_q"), 0.0))
        min_course_q = safe_float(class_cfg.get("min_course_q"), safe_float(global_cfg.get("min_course_q"), 0.0))
        max_area_ratio = safe_float(class_cfg.get("max_area_ratio"), safe_float(global_cfg.get("max_area_ratio"), 1.0))

        if score < min_score:
            continue
        if quality_q < min_quality_q:
            continue
        if course_q < min_course_q:
            continue
        if area_ratio > max_area_ratio:
            continue
        if xywhn[2] <= 0 or xywhn[3] <= 0:
            continue

        dets.append(
            CurriculumDet(
                class_name=class_name,
                class_id=class_id,
                xywhn=xywhn,
                box_xyxy=box_xyxy,
                score=score,
                quality_q=quality_q,
                course_q=course_q,
                support_prompt_count=support_prompt_count,
                support_prompt_ratio=support_prompt_ratio,
                prompt_hit_count_in_image=prompt_hit_count,
                box_dispersion=box_dispersion,
                teacher_risk_hint=teacher_risk_hint,
                selected_level=selected_level,
                area_ratio=area_ratio,
                local_evidence_q=local_evidence_q,
                drift=drift,
                meta_source=str(meta_path),
            )
        )

    return ImageRecord(
        image_path=str(image_path),
        rel_stem=rel_stem,
        width=image_w,
        height=image_h,
        detections=dets,
    )

def load_list_file(path_str: str) -> set[str]:
    if not path_str:
        return set()
    path = Path(path_str).expanduser().resolve()
    if not path.is_file():
        raise FileNotFoundError(f"列表文件不存在: {path}")
    values: set[str] = set()
    for line in path.read_text(encoding="utf-8").splitlines():
        s = line.strip()
        if not s or s.startswith("#"):
            continue
        values.add(Path(s).with_suffix("").as_posix())
    return values


def derive_group_key(rel_stem: str, split_cfg: Dict[str, Any]) -> str:
    mode = str(split_cfg.get("group_by", split_cfg.get("mode", "group"))).strip().lower()
    p = Path(rel_stem)
    parts = list(p.parts)

    if mode in {"image", "none", "per_image"}:
        return rel_stem

    if mode in {"parent", "group", "path_prefix"}:
        depth = max(1, safe_int(split_cfg.get("group_depth"), 1))
        if len(parts) > 1:
            return "/".join(parts[:depth])
        return rel_stem

    if mode in {"filename_prefix", "regex_prefix"}:
        stem = p.name
        pattern = str(split_cfg.get("prefix_regex", r"([_-]?(frame|img))?[_-]?\d{3,}$"))
        return re.sub(pattern, "", stem).strip("_-") or stem

    return rel_stem


def build_split_assignment(rel_stems: Sequence[str], split_cfg: Dict[str, Any]) -> Dict[str, str]:
    seed = safe_int(split_cfg.get("seed"), 3407)
    train_ratio = safe_float(split_cfg.get("train_ratio"), 0.9)
    train_ratio = max(0.0, min(1.0, train_ratio))

    val_list = load_list_file(str(split_cfg.get("val_list", "") or ""))
    train_list = load_list_file(str(split_cfg.get("train_list", "") or ""))

    if val_list & train_list:
        overlap = sorted(val_list & train_list)[:10]
        raise ValueError(f"train_list 和 val_list 有重叠: {overlap}")

    assignment: Dict[str, str] = {}
    unresolved: List[str] = []
    for rel_stem in rel_stems:
        if rel_stem in val_list:
            assignment[rel_stem] = "val"
        elif rel_stem in train_list:
            assignment[rel_stem] = "train"
        else:
            unresolved.append(rel_stem)

    if not unresolved:
        return assignment

    split_unit = str(split_cfg.get("split_unit", "group")).strip().lower()
    if split_unit in {"image", "per_image"}:
        group_map = {x: [x] for x in unresolved}
    else:
        group_map: Dict[str, List[str]] = defaultdict(list)
        for rel_stem in unresolved:
            group_key = derive_group_key(rel_stem, split_cfg)
            group_map[group_key].append(rel_stem)

    items = list(group_map.items())
    rnd = random.Random(seed)
    rnd.shuffle(items)

    total = sum(len(v) for _, v in items)
    target_train = int(round(total * train_ratio))
    train_count = 0

    for _group_key, members in items:
        split_name = "train" if train_count < target_train else "val"
        for rel_stem in members:
            assignment[rel_stem] = split_name
        if split_name == "train":
            train_count += len(members)

    return assignment

def load_stage_rules(cfg: Dict[str, Any]) -> Dict[str, StageRule]:
    stages_cfg = (((cfg.get("curriculum") or {}).get("stages")) or {})
    rules: Dict[str, StageRule] = {}
    for stage in ("stage1", "stage2", "stage3"):
        sc = stages_cfg.get(stage, {}) or {}
        rules[stage] = StageRule(
            name=stage,
            min_q=safe_float(sc.get("min_q"), 0.0),
            min_support_prompt_count=safe_int(sc.get("min_support_prompt_count"), 0),
            min_support_prompt_ratio=safe_float(sc.get("min_support_prompt_ratio"), 0.0),
            allow_high_risk=bool(sc.get("allow_high_risk", True)),
            include_empty_images=bool(sc.get("include_empty_images", False)),
        )
    return rules


def load_validation_rule(cfg: Dict[str, Any]) -> ValidationRule:
    vc = (cfg.get("validation") or {})
    rc = (vc.get("rule") or {})
    return ValidationRule(
        min_q=safe_float(rc.get("min_q"), 0.0),
        min_support_prompt_count=safe_int(rc.get("min_support_prompt_count"), 0),
        min_support_prompt_ratio=safe_float(rc.get("min_support_prompt_ratio"), 0.0),
        allow_high_risk=bool(rc.get("allow_high_risk", True)),
        include_empty_images=bool(vc.get("include_empty_images", False)),
    )


def compute_stage_thresholds(train_records: Sequence[ImageRecord], cfg: Dict[str, Any]) -> Dict[str, Dict[str, float]]:
    curriculum = cfg.get("curriculum", {}) or {}
    top_portions = curriculum.get("top_portions", {}) or {}
    min_floor_q = curriculum.get("min_floor_q", {}) or {}
    default_thresholds = curriculum.get("default_thresholds", {}) or {}
    min_class_samples_for_percentile = safe_int(curriculum.get("min_class_samples_for_percentile"), 10)
    stage_rules = load_stage_rules(cfg)
    class_names = list((cfg.get("classes") or {}).keys())

    class_qs: Dict[str, List[float]] = defaultdict(list)
    for rec in train_records:
        for det in rec.detections:
            class_qs[det.class_name].append(det.course_q)

    thresholds: Dict[str, Dict[str, float]] = {}
    for stage in ("stage1", "stage2", "stage3"):
        portion = safe_float(top_portions.get(stage), 1.0)
        portion = max(0.0, min(1.0, portion))
        floor_q = safe_float(min_floor_q.get(stage), 0.0)
        default_thr = safe_float(default_thresholds.get(stage), floor_q)
        stage_thr: Dict[str, float] = {}
        for class_name in class_names:
            qs = class_qs.get(class_name, [])
            if len(qs) >= min_class_samples_for_percentile and 0.0 < portion < 1.0:
                percentile_thr = quantile(qs, 1.0 - portion)
            else:
                percentile_thr = default_thr
            final_thr = max(
                stage_rules[stage].min_q,
                floor_q,
                percentile_thr,
            )
            stage_thr[class_name] = clamp01(final_thr)
        thresholds[stage] = stage_thr
    return thresholds

def det_pass_stage(det: CurriculumDet, stage_rule: StageRule, class_threshold: float) -> bool:
    if det.course_q < class_threshold:
        return False
    if det.support_prompt_count < stage_rule.min_support_prompt_count:
        return False
    if det.support_prompt_ratio < stage_rule.min_support_prompt_ratio:
        return False
    if (not stage_rule.allow_high_risk) and det.teacher_risk_hint == "high":
        return False
    return True


def det_pass_val(det: CurriculumDet, val_rule: ValidationRule) -> bool:
    if det.course_q < val_rule.min_q:
        return False
    if det.support_prompt_count < val_rule.min_support_prompt_count:
        return False
    if det.support_prompt_ratio < val_rule.min_support_prompt_ratio:
        return False
    if (not val_rule.allow_high_risk) and det.teacher_risk_hint == "high":
        return False
    return True


def take_top_per_class(dets: Sequence[CurriculumDet], max_boxes_per_image_per_class: int) -> List[CurriculumDet]:
    if max_boxes_per_image_per_class <= 0:
        return list(dets)
    bucket: Dict[str, List[CurriculumDet]] = defaultdict(list)
    for det in dets:
        bucket[det.class_name].append(det)
    out: List[CurriculumDet] = []
    for _class_name, values in bucket.items():
        values = sorted(values, key=lambda x: (x.course_q, x.score), reverse=True)
        out.extend(values[:max_boxes_per_image_per_class])
    return out


def write_yolo_label(label_path: Path, dets: Sequence[CurriculumDet]) -> None:
    ensure_parent(label_path)
    lines: List[str] = []
    for det in dets:
        x, y, w, h = det.xywhn
        lines.append(f"{det.class_id} {x:.6f} {y:.6f} {w:.6f} {h:.6f}")
    label_path.write_text("\n".join(lines) + ("\n" if lines else ""), encoding="utf-8")


def export_split_dataset(
    records: Sequence[ImageRecord],
    out_dir: Path,
    image_link_mode: str,
    det_selector,
    include_empty_images: bool,
    max_boxes_per_image_per_class: int,
) -> Dict[str, Any]:
    img_out = out_dir / "images"
    lab_out = out_dir / "labels"
    if out_dir.exists():
        shutil.rmtree(out_dir)
    img_out.mkdir(parents=True, exist_ok=True)
    lab_out.mkdir(parents=True, exist_ok=True)

    image_count = 0
    labeled_image_count = 0
    empty_image_count = 0
    total_boxes = 0
    class_counter: Counter[str] = Counter()

    for rec in records:
        dets = det_selector(rec)
        dets = take_top_per_class(dets, max_boxes_per_image_per_class)
        keep_image = bool(dets) or include_empty_images
        if not keep_image:
            continue

        src_img = Path(rec.image_path)
        dst_img = img_out / f"{rec.rel_stem}{src_img.suffix.lower()}"
        dst_lab = lab_out / f"{rec.rel_stem}.txt"
        maybe_symlink_or_copy(src_img, dst_img, mode=image_link_mode)
        write_yolo_label(dst_lab, dets)

        image_count += 1
        total_boxes += len(dets)
        if dets:
            labeled_image_count += 1
            for det in dets:
                class_counter[det.class_name] += 1
        else:
            empty_image_count += 1

    return {
        "images": image_count,
        "labeled_images": labeled_image_count,
        "empty_images": empty_image_count,
        "boxes": total_boxes,
        "boxes_by_class": dict(class_counter),
    }


def build_dataset_yaml(stage_dir: Path, names_map: Dict[int, str], val_images_dir: Path) -> Dict[str, Any]:
    names = [name for _, name in sorted(names_map.items(), key=lambda kv: kv[0])]
    data = {
        "path": str(stage_dir.resolve()),
        "train": str((stage_dir / "train" / "images").resolve()),
        "val": str(val_images_dir.resolve()),
        "names": names,
        "nc": len(names),
    }
    dump_yaml(stage_dir / "dataset.yaml", data)
    return data

def summarize_records(records: Sequence[ImageRecord]) -> Dict[str, Any]:
    image_count = len(records)
    labeled_images = sum(1 for r in records if r.detections)
    empty_images = image_count - labeled_images
    box_count = sum(len(r.detections) for r in records)
    by_class: Counter[str] = Counter()
    q_values: Dict[str, List[float]] = defaultdict(list)
    for r in records:
        for d in r.detections:
            by_class[d.class_name] += 1
            q_values[d.class_name].append(d.course_q)
    q_summary = {}
    for k, vals in q_values.items():
        q_summary[k] = {
            "count": len(vals),
            "min": min(vals),
            "p25": quantile(vals, 0.25),
            "p50": quantile(vals, 0.50),
            "p75": quantile(vals, 0.75),
            "max": max(vals),
            "mean": sum(vals) / max(1, len(vals)),
        }
    return {
        "images": image_count,
        "labeled_images": labeled_images,
        "empty_images": empty_images,
        "boxes": box_count,
        "boxes_by_class": dict(by_class),
        "course_q_by_class": q_summary,
    }


def build_split_manifest(
    assignment: Dict[str, str],
    rel_stems: Sequence[str],
    split_cfg: Dict[str, Any],
) -> Dict[str, Any]:
    out = {
        "split_config": split_cfg,
        "counts": {
            "train": sum(1 for x in assignment.values() if x == "train"),
            "val": sum(1 for x in assignment.values() if x == "val"),
        },
        "items": [],
    }
    for rel_stem in sorted(rel_stems):
        out["items"].append(
            {
                "rel_stem": rel_stem,
                "split": assignment[rel_stem],
                "group_key": derive_group_key(rel_stem, split_cfg),
            }
        )
    return out

def main() -> None:
    args = parse_args()
    cfg = load_json(Path(args.config).expanduser().resolve())

    paths_cfg = cfg.get("paths", {}) or {}
    image_dir = Path(paths_cfg["image_dir"]).expanduser().resolve()
    meta_dir = Path(paths_cfg["meta_dir"]).expanduser().resolve()
    output_root = Path(paths_cfg["output_root"]).expanduser().resolve()
    recursive = bool(paths_cfg.get("recursive", True))
    image_link_mode = str(paths_cfg.get("image_link_mode", "symlink"))

    name_to_id, id_to_name = load_names_map(cfg)

    image_map = discover_images(image_dir, recursive=recursive)
    meta_map = discover_meta_files(meta_dir, recursive=recursive)
    common_rel_stems = sorted(set(image_map.keys()) & set(meta_map.keys()))
    if not common_rel_stems:
        raise RuntimeError("没有找到 image/meta 对齐的样本，请检查 image_dir 和 meta_dir")

    split_cfg = cfg.get("split", {}) or {}
    assignment = build_split_assignment(common_rel_stems, split_cfg)
    manifest = build_split_manifest(assignment, common_rel_stems, split_cfg)
    dump_json(output_root / "split_manifest.json", manifest)

    print(f"[split] total={len(common_rel_stems)} train={manifest['counts']['train']} val={manifest['counts']['val']}")

    all_records: Dict[str, ImageRecord] = {}
    for rel_stem in common_rel_stems:
        rec = parse_one_meta(meta_map[rel_stem], image_map[rel_stem], rel_stem, cfg, name_to_id)
        if rec is None:
            continue
        all_records[rel_stem] = rec

    train_records = [all_records[x] for x in common_rel_stems if x in all_records and assignment[x] == "train"]
    val_records = [all_records[x] for x in common_rel_stems if x in all_records and assignment[x] == "val"]
    if not train_records:
        raise RuntimeError("train_pool 为空，无法继续")
    if not val_records:
        raise RuntimeError("val_holdout 为空，无法继续")

    dump_json(output_root / "train_pool_summary.json", summarize_records(train_records))
    dump_json(output_root / "val_pool_summary.json", summarize_records(val_records))

    stage_rules = load_stage_rules(cfg)
    val_rule = load_validation_rule(cfg)
    thresholds = compute_stage_thresholds(train_records, cfg)
    dump_json(output_root / "curriculum_thresholds_train_only.json", thresholds)

    curriculum_cfg = cfg.get("curriculum", {}) or {}
    max_boxes_per_image_per_class = safe_int(curriculum_cfg.get("max_boxes_per_image_per_class"), 0)

    # fixed holdout val exported once; shared by all stages
    holdout_dir = output_root / "holdout_val"
    holdout_stats = export_split_dataset(
        records=val_records,
        out_dir=holdout_dir,
        image_link_mode=image_link_mode,

        include_empty_images=val_rule.include_empty_images,
        max_boxes_per_image_per_class=max_boxes_per_image_per_class,
    )
    dump_json(output_root / "holdout_val" / "stats.json", holdout_stats)

    # stage train sets
    stage_reports: Dict[str, Any] = {
        "design": {
            "no_leakage": True,
            "rule": "先 split 成 train_pool / val_holdout，再只用 train_pool 计算课程阈值；val_holdout 不参与 percentile/threshold/top_portion 估计。",
            "shared_val_dir": str((holdout_dir / 'images').resolve()),
        },
        "train_only_thresholds": thresholds,
        "holdout_val_stats": holdout_stats,
        "stages": {},
    }

    for stage in ("stage1", "stage2", "stage3"):
        stage_dir = output_root / stage
        stage_train_dir = stage_dir / "train"
        thr_map = thresholds[stage]
        stage_rule = stage_rules[stage]
        train_stats = export_split_dataset(
            records=train_records,
            out_dir=stage_train_dir,
            image_link_mode=image_link_mode,
            det_selector=lambda rec, s=stage, r=stage_rule: [
                d for d in rec.detections if det_pass_stage(d, r, thresholds[s].get(d.class_name, r.min_q))
            ],
            include_empty_images=stage_rule.include_empty_images,
            max_boxes_per_image_per_class=max_boxes_per_image_per_class,
        )
        data_yaml = build_dataset_yaml(stage_dir, id_to_name, holdout_dir / "images")
        report = {
            "stage": stage,
            "rule": asdict(stage_rule),
            "thresholds_by_class": thr_map,
            "train_stats": train_stats,
            "val_stats": holdout_stats,
            "dataset_yaml": data_yaml,
        }
        dump_json(stage_dir / "report.json", report)
        stage_reports["stages"][stage] = report
        print(f"[{stage}] train_images={train_stats['images']} boxes={train_stats['boxes']} val_images={holdout_stats['images']}")

    dump_json(output_root / "curriculum_build_report.json", stage_reports)

    print("=" * 80)
    print("构建完成（无验证泄露版）")
    print(f"output_root      : {output_root}")
    print(f"split_manifest   : {output_root / 'split_manifest.json'}")
    print(f"train thresholds : {output_root / 'curriculum_thresholds_train_only.json'}")
    print(f"shared val dir   : {holdout_dir / 'images'}")
    print("说明: 三个 stage 的 dataset.yaml 均指向同一个 holdout_val，不再在 stage 内部重新切 val。")


if __name__ == "__main__":
    main()

def derive_route(pred_count: int, ref_count: int, matched_count: int, mean_iou: float, keep_thr: float, sam_thr: float, review_iou_thr: float) -> str:
    if pred_count == 0 and ref_count == 0:
        return "keep_yolo"
    if pred_count == 0 and ref_count > 0:
        return "send_sam"
    if pred_count > 0 and ref_count == 0:
        return "send_vlm_direct"

    coverage = matched_count / max(1, ref_count)
    precision = matched_count / max(1, pred_count)
    consistency = min(coverage, precision)

    if consistency >= keep_thr and mean_iou >= review_iou_thr:
        return "keep_yolo"
    if coverage >= sam_thr or matched_count > 0:
        return "send_sam"
    return "send_vlm_direct"


def load_teacher_meta(ref_meta_dir: Optional[Path], sample_key: str) -> Optional[Dict[str, Any]]:
    if not ref_meta_dir:
        return None
    key = sanitize_key(sample_key)
    candidates = [
        ref_meta_dir / f"{key}.json",
        ref_meta_dir / f"{Path(key).name}.json",
        ref_meta_dir / key / "meta.json",
    ]
    for p in candidates:
        if p.exists():
            try:
                return json.loads(p.read_text(encoding="utf-8"))
            except Exception:
                return None
    return None


def meta_for_ref_box(meta: Optional[Dict[str, Any]], ref_idx: int) -> Dict[str, Any]:
    if not isinstance(meta, dict):
        return {}
    # Common shapes supported by GroundingDINO exporters.
    for key in ["boxes", "detections", "items", "labels", "instances"]:
        arr = meta.get(key)
        if isinstance(arr, list) and 0 <= ref_idx < len(arr) and isinstance(arr[ref_idx], dict):
            return arr[ref_idx]
    # Some exporters keep per-box metadata flat only for single-box JSON.
    if ref_idx == 0:
        return meta
    return {}


def support_risk(box_meta: Dict[str, Any], args: argparse.Namespace) -> Tuple[float, List[str]]:
    risk = 0.0
    tags: List[str] = []

    ratio = safe_float(box_meta.get("support_prompt_ratio"), -1.0)
    if ratio >= 0 and ratio < args.dino_low_support_ratio_thr:
        risk += 0.35
        tags.append("low_support_ratio")

    count = safe_int(box_meta.get("support_prompt_count"), -1)
    if count >= 0 and count < 1:
        risk += 0.20
        tags.append("zero_support_prompt")

    cluster_size = safe_int(box_meta.get("cluster_size"), -1)
    if cluster_size >= 0 and cluster_size < args.dino_min_cluster_size:
        risk += 0.25
        tags.append("small_cluster")

    score_mean = safe_float(box_meta.get("score_mean"), -1.0)
    score_max = safe_float(box_meta.get("score_max"), -1.0)
    score = score_mean if score_mean >= 0 else score_max
    if score >= 0 and score < args.dino_low_score_thr:
        risk += 0.25
        tags.append("low_teacher_score")

    dispersion = safe_float(box_meta.get("box_dispersion"), -1.0)
    if dispersion >= 0 and dispersion > args.dino_high_dispersion_thr:
        risk += 0.30
        tags.append("high_box_dispersion")

    hint = str(box_meta.get("teacher_risk_hint", "")).lower()
    if hint and hint not in {"none", "ok", "low", "false", "0"}:
        risk += 0.30
        tags.append(f"teacher_hint:{hint[:24]}")

    return min(1.0, risk), tags


def geometry_risk(box: Box) -> Tuple[float, List[str]]:
    risk = 0.0
    tags: List[str] = []
    area = max(0.0, box.w * box.h)
    if area < 0.0003:
        risk += 0.20
        tags.append("tiny_box")
    if area > 0.65:
        risk += 0.25
        tags.append("huge_box")
    if box.xc - box.w / 2 <= 0.005 or box.yc - box.h / 2 <= 0.005 or box.xc + box.w / 2 >= 0.995 or box.yc + box.h / 2 >= 0.995:
        risk += 0.15
        tags.append("touch_border")
    ratio = max(box.w, box.h) / max(1e-6, min(box.w, box.h))
    if ratio > 10:
        risk += 0.15
        tags.append("extreme_aspect")
    return min(1.0, risk), tags


def combined_risk(pred: Box, ref_meta: Dict[str, Any], args: argparse.Namespace) -> Tuple[float, str]:
    g, gt = geometry_risk(pred)
    s, st = support_risk(ref_meta, args)
    score = min(1.0, max(g, s, 0.55 * g + 0.75 * s))
    tags = gt + st
    return score, ";".join(tags) if tags else "low_risk"


def predict_yolo_labels(args: argparse.Namespace, image_paths: List[Path], image_root: Path, pred_label_root: Path) -> Dict[str, Path]:
    try:
        from ultralytics import YOLO
    except Exception as e:
        raise ImportError("无法导入 ultralytics.YOLO，请先安装 ultralytics。") from e

    model = YOLO(args.weights)
    pred_map: Dict[str, Path] = {}

    for img_path in image_paths:
        sample_key = key_from_image(img_path, image_root)
        label_path = pred_label_root / f"{sample_key}.txt"
        ensure_dir(label_path.parent)
        try:
            results = model.predict(
                source=str(img_path),
                imgsz=args.imgsz,
                conf=args.conf,
                iou=args.iou,
                device=args.device,
                verbose=False,
            )
        except TypeError:
            results = model.predict(str(img_path), imgsz=args.imgsz, conf=args.conf, iou=args.iou, verbose=False)

        boxes: List[Box] = []
        if results:
            r = results[0]
            if getattr(r, "boxes", None) is not None and len(r.boxes) > 0:
                im = Image.open(img_path)
                img_w, img_h = im.size
                xyxy = r.boxes.xyxy.detach().cpu().numpy()
                cls = r.boxes.cls.detach().cpu().numpy()
                conf = r.boxes.conf.detach().cpu().numpy() if getattr(r.boxes, "conf", None) is not None else [0.0] * len(cls)
                for b, c, cf in zip(xyxy, cls, conf):
                    cid = int(round(float(c)))
                    if cid not in ID_TO_CLASS:
                        continue
                    xc, yc, bw, bh = xyxy_to_norm(float(b[0]), float(b[1]), float(b[2]), float(b[3]), img_w, img_h)
                    if bw <= 0 or bh <= 0:
                        continue
                    boxes.append(Box(cid, xc, yc, bw, bh, source="yolo", conf=float(cf), extra={}))
        write_yolo_labels(label_path, boxes, with_conf=args.save_conf_in_pred_labels)
        pred_map[sanitize_key(sample_key)] = label_path
    return pred_map


def main() -> None:
    args = parse_args()
    random.seed(args.seed)

    image_root = Path(args.images)
    ref_label_root = Path(args.ref_labels)

    print(f"[INFO] running YOLO and writing pred labels to: {pred_label_root}")
    pred_map = predict_yolo_labels(args, image_paths, image_root, pred_label_root)

    manifest_rows: List[Dict[str, Any]] = []
    vlm_rows: List[Dict[str, Any]] = []
    route_buckets: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
    route_counter = Counter()
    teacher_meta_found_images = 0

    for img_path in image_paths:
        sample_key = key_from_image(img_path, image_root)
        safe_key = sanitize_key(sample_key)
        with Image.open(img_path) as im:
            img_w, img_h = im.size

        pred_path = pred_map.get(safe_key, pred_label_root / f"{safe_key}.txt")
        ref_path = ref_label_root / f"{safe_key}.txt"
        if not ref_path.exists():
            alt = ref_label_root / f"{Path(safe_key).name}.txt"
            ref_path = alt if alt.exists() else ref_path

        preds = load_yolo_labels(pred_path, source="yolo")
        refs = load_yolo_labels(ref_path, source="ref")
        matches = greedy_match(preds, refs, img_w, img_h, iou_thr=args.match_iou)
        mean_iou = sum(m.iou for m in matches) / max(1, len(matches))
        route = derive_route(len(preds), len(refs), len(matches), mean_iou, args.keep_thr, args.sam_thr, args.review_iou_thr)
        route_counter[route] += 1

        meta = load_teacher_meta(ref_meta_dir, safe_key) if ref_meta_dir else None
        if meta is not None:
            teacher_meta_found_images += 1

        matched_pred_indices = {m.pred_idx for m in matches}
        matched_ref_by_pred = {m.pred_idx: m.ref_idx for m in matches}
        matched_ref_indices = {m.ref_idx for m in matches}
        coverage = len(matches) / max(1, len(refs)) if refs else 1.0
        precision = len(matches) / max(1, len(preds)) if preds else 1.0
        match_ratio = min(coverage, precision)

        row = {
            "sample_key": safe_key,
            "image_path": str(img_path),
            "image_name": img_path.name,
            "pred_label_path": str(pred_path),
            "ref_label_path": str(ref_path) if ref_path.exists() else "",
            "route": route,
            "stage1_route": route,
            "pred_count": len(preds),
            "ref_count": len(refs),
            "matched_count": len(matches),
            "match_ratio": round(match_ratio, 6),
            "coverage": round(coverage, 6),
            "precision": round(precision, 6),
            "mean_iou": round(mean_iou, 6),
            "unmatched_pred_count": max(0, len(preds) - len(matched_pred_indices)),
            "unmatched_ref_count": max(0, len(refs) - len(matched_ref_indices)),
            "ref_meta_json_path": str(ref_meta_dir / f"{safe_key}.json") if ref_meta_dir else "",
            "teacher_meta_found": int(meta is not None),
            "next_route_if_sam_has_boxes": "send_vlm_after_sam",
            "next_route_if_sam_empty": "hold",
        }
        manifest_rows.append(row)
        route_buckets[route].append(row)

        if args.export_vlm_consensus_candidates:
            if route == "send_vlm_direct":
                for p in preds:
                    risk, tags = combined_risk(p, {}, args)
                    vlm_rows.append(candidate_row_from_box(safe_key, img_path, img_w, img_h, p, "yolo", "direct_vlm", max(risk, 0.50), tags or "direct_vlm", args.vlm_crop_expand_ratio))
            elif route == "keep_yolo":
                for pi, p in enumerate(preds):
                    ref_meta = meta_for_ref_box(meta, matched_ref_by_pred.get(pi, -1)) if pi in matched_ref_by_pred else {}
                    risk, tags = combined_risk(p, ref_meta, args)
                    audit = random.random() < max(0.0, min(1.0, args.audit_sample_rate))
                    if risk >= args.audit_risk_thr or audit:
                        stage = "consensus_keep_audit" if audit and risk < args.audit_risk_thr else "consensus_keep"
                        vlm_rows.append(candidate_row_from_box(safe_key, img_path, img_w, img_h, p, "yolo", stage, risk, tags, args.vlm_crop_expand_ratio))

    manifest_header = [
        "sample_key", "image_path", "image_name", "pred_label_path", "ref_label_path", "route", "stage1_route",
        "pred_count", "ref_count", "matched_count", "match_ratio", "coverage", "precision", "mean_iou",
        "unmatched_pred_count", "unmatched_ref_count", "ref_meta_json_path", "teacher_meta_found",
        "next_route_if_sam_has_boxes", "next_route_if_sam_empty",
    ]
    write_csv(report_dir / "compare_manifest.csv", manifest_rows, manifest_header)
    if args.export_routes:
        for name, rows in route_buckets.items():
            write_csv(report_dir / f"compare_manifest_{name}.csv", rows, manifest_header)

    vlm_header = [
        "sample_key", "image_path", "box_uid", "source", "review_stage", "cls_id", "cls_name",
        "x1", "y1", "x2", "y2", "xc", "yc", "w", "h",
        "box_risk_score", "risk_tags", "touch_border", "crop_expand_ratio",
    ]
    if args.export_vlm_consensus_candidates:
        write_csv(report_dir / "vlm_consensus_candidates.csv", vlm_rows, vlm_header)

    summary = {
        "images": len(image_paths),
        "routes": dict(route_counter),
        "teacher_meta_found_images": teacher_meta_found_images,
        "outputs": {
            "pred_labels": str(pred_label_root),
            "compare_manifest": str(report_dir / "compare_manifest.csv"),
            "vlm_consensus_candidates": str(report_dir / "vlm_consensus_candidates.csv") if args.export_vlm_consensus_candidates else None,
        },
        "params": vars(args),
        "policy": "YOLO inference is performed inside compare; pred_labels are generated under out/pred_labels.",
    }
    write_json(report_dir / "compare_summary.json", summary)
    print("=" * 80)
    print("[DONE] compare-routing finished")
    print(json.dumps(summary, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()

            return default

    def make_box_uid(sample_key: str, source: str, cls_id: int, xc: float, yc: float, w: float, h: float) -> str:
        raw = f"{sanitize_key(sample_key)}|{source}|{cls_id}|{xc:.6f}|{yc:.6f}|{w:.6f}|{h:.6f}"
        return hashlib.md5(raw.encode("utf-8")).hexdigest()

    def write_csv(path: Path | str, rows: List[Dict[str, Any]]) -> None:
        path = Path(path)
        ensure_dir(path.parent)
        if not rows:
            with open(path, "w", encoding="utf-8", newline="") as f:
                pass
            return
        fieldnames: List[str] = []
        seen = set()
        for row in rows:
            for k in row.keys():
                if k not in seen:
                    seen.add(k)
                    fieldnames.append(k)
        with open(path, "w", encoding="utf-8", newline="") as f:
            writer = csv.DictWriter(f, fieldnames=fieldnames)
            writer.writeheader()
            writer.writerows(rows)

    def box_area(box: Iterable[float]) -> float:
        x1, y1, x2, y2 = [float(v) for v in box]
        return max(0.0, x2 - x1) * max(0.0, y2 - y1)

    def expand_box(box: Iterable[float], img_w: int, img_h: int, ratio: float = 0.0) -> List[float]:
        x1, y1, x2, y2 = [float(v) for v in box]
        if ratio <= 0:
            return [x1, y1, x2, y2]
        w = x2 - x1
        h = y2 - y1
        dw = w * ratio / 2.0
        dh = h * ratio / 2.0
        return [
            max(0.0, x1 - dw),
            max(0.0, y1 - dh),
            min(float(img_w - 1), x2 + dw),
            min(float(img_h - 1), y2 + dh),
        ]

    def rank_score(cls_name: str, conf: float, area: float, img_area: float) -> float:
        area_ratio = area / max(1.0, img_area)
        if cls_name == "fire":
            area_bonus = min(0.25, area_ratio / 0.01 * 0.25)
            return float(conf + area_bonus)
        area_bonus = min(0.10, area_ratio / 0.04 * 0.10)
        return float(conf + area_bonus)

    def postprocess_v2(
        proposals: List[Dict[str, Any]],
        params: Dict[str, Any],
        img_w: int,
        img_h: int,
    ) -> Tuple[List[Dict[str, Any]], Dict[str, int]]:
        by_cls: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
        for p in proposals:
            by_cls[str(p.get("cls_name"))].append(p)

        out: List[Dict[str, Any]] = []
        post_stats: Dict[str, int] = {
            "fire_before_merge": 0,
            "fire_after_merge": 0,
            "fire_after_nms": 0,
            "fire_after_iof": 0,
            "fire_after_topk": 0,
            "smoke_before_merge": 0,
            "smoke_after_merge": 0,
            "smoke_after_nms": 0,
            "smoke_after_iof": 0,
            "smoke_after_topk": 0,
        }

        for cls_name, arr in by_cls.items():
            topk = int(params.get(f"{cls_name}_topk", 4))
            post_stats[f"{cls_name}_before_merge"] = len(arr)
            arr = sorted(arr, key=lambda x: float(x.get("rank_score", 0.0)), reverse=True)
            picked = arr[:topk]
            post_stats[f"{cls_name}_after_merge"] = len(picked)
            post_stats[f"{cls_name}_after_nms"] = len(picked)
            post_stats[f"{cls_name}_after_iof"] = len(picked)
            post_stats[f"{cls_name}_after_topk"] = len(picked)
            out.extend(picked)

        return out, post_stats

def save_yolo_labels(label_path: Path, props: List[Dict[str, Any]], w: int, h: int) -> int:
    lines: List[str] = []
    for p in props:
        if not isinstance(p, dict):
            continue
        cls_name = str(p.get("cls_name", ""))
        if cls_name not in CLASS_TO_ID:
            continue
        if "xyxy" not in p:
            continue

        x1, y1, x2, y2 = [float(v) for v in p["xyxy"]]
        x1 = max(0.0, min(float(w - 1), x1))
        y1 = max(0.0, min(float(h - 1), y1))
        x2 = max(0.0, min(float(w - 1), x2))
        y2 = max(0.0, min(float(h - 1), y2))
        bw = max(0.0, x2 - x1)
        bh = max(0.0, y2 - y1)
        if bw <= 1 or bh <= 1:
            continue

        xc = (x1 + x2) / 2.0 / max(1, w)
        yc = (y1 + y2) / 2.0 / max(1, h)
        wn = bw / max(1, w)
        hn = bh / max(1, h)
        lines.append(f"{CLASS_TO_ID[cls_name]} {xc:.6f} {yc:.6f} {wn:.6f} {hn:.6f}")

    ensure_dir(label_path.parent)
    label_path.write_text("\n".join(lines), encoding="utf-8")
    return len(lines)


def save_payload(
    npz_path: Path,
    masks: np.ndarray,
    confs: np.ndarray,
    clss: np.ndarray,
    names: Any,
    h: int,
    w: int,
) -> None:
    ensure_dir(npz_path.parent)
    np.savez(
        npz_path,
        masks=masks.astype(np.uint8),
        confs=confs.astype(np.float32),
        clss=clss.astype(np.int32),
        names_json=np.array(json.dumps(names, ensure_ascii=False), dtype=object),
        H=np.array([h], dtype=np.int32),
        W=np.array([w], dtype=np.int32),
    )


def normalize_names(names: Any) -> Dict[int, str]:
    if isinstance(names, dict):
        return {int(k): str(v) for k, v in names.items()}
    if isinstance(names, (list, tuple)):
        return {i: str(v) for i, v in enumerate(names)}
    return {}


def load_payload(npz_path: Path) -> Dict[str, Any]:
    data = np.load(npz_path, allow_pickle=True)
    names = normalize_names(json.loads(str(data["names_json"])))
    return {
        "masks": data["masks"],
        "confs": data["confs"],
        "clss": data["clss"],
        "names": names,
        "H": int(data["H"][0]),
        "W": int(data["W"][0]),
    }


def empty_payload(h: int, w: int) -> Dict[str, Any]:
    return {
        "masks": np.empty((0, h, w), dtype=np.uint8),
        "confs": np.empty((0,), dtype=np.float32),
        "clss": np.empty((0,), dtype=np.int32),
        "names": {},
        "H": int(h),
        "W": int(w),
    }


def predictor_set_image(predictor: Any, image_path: str) -> bool:
    if hasattr(predictor, "set_image"):
        try:
            predictor.set_image(image_path)
            return True
        except Exception:
            return False
    return False


def predictor_reset_image(predictor: Any) -> None:
    for fn_name in ["reset_image", "reset", "clear_image"]:
        if hasattr(predictor, fn_name):
            try:
                getattr(predictor, fn_name)()
                return
            except Exception:
                pass


def predictor_call(
    predictor: Any,
    image_path: str,
    prompt: str,
    image_already_set: bool = False,
):
    if image_already_set:
        try:
            return predictor(text=[prompt])
        except Exception:
            pass

    try:
        return predictor(source=image_path, text=[prompt])
    except TypeError:
        pass
    except Exception:
        pass

    try:
        return predictor(image_path, text=[prompt])
    except TypeError:
        pass
    except Exception:
        pass

    if not image_already_set:
        try:
            if predictor_set_image(predictor, image_path):
                return predictor(text=[prompt])
        except Exception:
            pass

    return predictor(text=[prompt])


def maybe_empty_cache() -> None:
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
    gc.collect()


def is_no_space_error(e: Exception) -> bool:
    return getattr(e, "errno", None) == 28 or "No space left on device" in str(e)


def safe_unlink(path: Path) -> None:
    try:
        if path.exists():
            path.unlink()
    except Exception:
        pass


def trim_raw_masks_per_class(
    masks: np.ndarray,
    confs: np.ndarray,
    clss: np.ndarray,
    max_raw_masks_per_class: int,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    if len(confs) == 0:
        return masks, confs, clss

    if max_raw_masks_per_class is None or int(max_raw_masks_per_class) <= 0:
        return masks, confs, clss

    keep_idx: List[int] = []
    for cls_id in np.unique(clss):
        idx = np.where(clss == cls_id)[0]
        if len(idx) <= max_raw_masks_per_class:
            keep_idx.extend(idx.tolist())
            continue
        order = np.argsort(confs[idx])[::-1]
        picked = idx[order[:max_raw_masks_per_class]]
        keep_idx.extend(picked.tolist())

    if not keep_idx:
        return (
            np.empty((0,) + masks.shape[1:], dtype=masks.dtype),
            np.empty((0,), dtype=confs.dtype),
            np.empty((0,), dtype=clss.dtype),
        )

    keep_idx = sorted(set(keep_idx))
    return masks[keep_idx], confs[keep_idx], clss[keep_idx]


def results_to_payload(
    results: Any,
    h: int,
    w: int,
    max_raw_masks_per_class: int,
) -> Dict[str, Any]:
    masks_list: List[np.ndarray] = []
    confs_list: List[np.ndarray] = []
    clss_list: List[np.ndarray] = []
    names: Any = {}

    for result in results:
        boxes = getattr(result, "boxes", None)
        masks_obj = getattr(result, "masks", None)
        if boxes is None or len(boxes) == 0 or masks_obj is None:
            continue

        masks = masks_obj.data.detach().cpu().numpy().astype(np.uint8)
        confs = boxes.conf.detach().cpu().numpy().astype(np.float32)
        clss = boxes.cls.detach().cpu().numpy().astype(np.int32)

        if len(confs) > 0:
            masks, confs, clss = trim_raw_masks_per_class(
                masks=masks,
                confs=confs,
                clss=clss,
                max_raw_masks_per_class=max_raw_masks_per_class,
            )
            if len(confs) > 0:
                masks_list.append(masks)
                confs_list.append(confs)
                clss_list.append(clss)

        names = getattr(result, "names", {}) or names

        try:
            del result.masks
        except Exception:
            pass
        try:
            del result.boxes
        except Exception:
            pass

    if not confs_list:
        return empty_payload(h, w)

    return {
        "masks": np.concatenate(masks_list, axis=0).astype(np.uint8),
        "confs": np.concatenate(confs_list, axis=0).astype(np.float32),
        "clss": np.concatenate(clss_list, axis=0).astype(np.int32),
        "names": normalize_names(names),
        "H": int(h),
        "W": int(w),
    }


def build_payload_for_prompt(
    predictor: Any,
    image_path: str,
    prompt: str,
    npz_path: Optional[Path],
    force_recache: bool,
    use_disk_cache: bool,
    max_raw_masks_per_class: int,
    image_already_set: bool,
    image_hw: Optional[Tuple[int, int]] = None,
) -> Dict[str, Any]:
    if use_disk_cache and npz_path is not None and npz_path.exists() and not force_recache:
        maybe_empty_cache()
        return load_payload(npz_path)

    if image_hw is None:
        img_bgr = cv2.imread(image_path)
        if img_bgr is None:
            raise RuntimeError(f"无法读取图片: {image_path}")
        h, w = img_bgr.shape[:2]
    else:
        h, w = int(image_hw[0]), int(image_hw[1])

    payload = empty_payload(h, w)

    try:
        results = predictor_call(
            predictor=predictor,
            image_path=image_path,
            prompt=prompt,
            image_already_set=image_already_set,
        )
        payload = results_to_payload(
            results=results,
            h=h,
            w=w,
            max_raw_masks_per_class=max_raw_masks_per_class,
        )
        try:
            del results
        except Exception:
            pass

        if use_disk_cache and npz_path is not None:
            try:
                save_payload(
                    npz_path=npz_path,
                    masks=payload["masks"],
                    confs=payload["confs"],
                    clss=payload["clss"],
                    names=payload["names"],
                    h=payload["H"],
                    w=payload["W"],
                )
                maybe_empty_cache()
                return load_payload(npz_path)
            except OSError as e:
                safe_unlink(npz_path)
                if is_no_space_error(e):
                    print(f"[WARN] cache save skipped (disk full): {Path(image_path).name}, prompt={prompt}")
                    return payload
                raise

        return payload

    except Exception as e:
        print(f"[WARN] predictor failed: {Path(image_path).name}, prompt={prompt}, err={e}")
        if "out of memory" in str(e).lower():
            maybe_empty_cache()

        payload = empty_payload(h, w)

        if use_disk_cache and npz_path is not None:
            try:
                save_payload(
                    npz_path=npz_path,
                    masks=payload["masks"],
                    confs=payload["confs"],
                    clss=payload["clss"],
                    names=payload["names"],
                    h=payload["H"],
                    w=payload["W"],
                )
                maybe_empty_cache()
                return load_payload(npz_path)
            except OSError as save_e:
                safe_unlink(npz_path)
                if is_no_space_error(save_e):
                    print(f"[WARN] empty cache save skipped (disk full): {Path(image_path).name}, prompt={prompt}")
                    return payload
                raise

        return payload

    finally:
        maybe_empty_cache()


def build_region_proposals_from_payload(
    payload: Dict[str, Any],
    params: Dict[str, Any],
    prompt_hint: Optional[str] = None,
) -> Tuple[List[Dict[str, Any]], Dict[str, int]]:
    debug_stats: Dict[str, int] = defaultdict(int)

    masks = payload["masks"]
    confs = payload["confs"]
    clss = payload["clss"]
    names = payload["names"]
    h, w = payload["H"], payload["W"]

    if len(confs) == 0 or masks.size == 0:
        return [], dict(debug_stats)

    img_area = h * w
    proposals: List[Dict[str, Any]] = []

    for cls_id in np.unique(clss):
        cls_name = names.get(int(cls_id), None)
        if cls_name not in ["fire", "smoke"] and prompt_hint in ["fire", "smoke"]:
            cls_name = prompt_hint
        if cls_name not in ["fire", "smoke"]:
            continue

        conf_thr = float(params.get(f"{cls_name}_conf", 0.5))
        min_area = int(params.get(f"{cls_name}_min_area", 100))
        open_k = int(params.get(f"{cls_name}_open", 1))
        close_k = int(params.get(f"{cls_name}_close", 1))
        dilate_k = int(params.get(f"{cls_name}_dilate", 1))
        box_pad = int(params.get(f"{cls_name}_box_pad", 0))
        expand_ratio = float(params.get(f"{cls_name}_expand_ratio", 0.0))

        raw_idx = np.where(clss == cls_id)[0]
        debug_stats[f"{cls_name}_raw_masks"] += int(len(raw_idx))

        idx = np.where((clss == cls_id) & (confs >= conf_thr))[0]
        debug_stats[f"{cls_name}_after_conf"] += int(len(idx))
        if len(idx) == 0:
            continue

        union_mask = np.zeros((h, w), dtype=np.uint8)
        conf_mask = np.zeros((h, w), dtype=np.float32)

        for i in idx:
            m = masks[i]
            union_mask |= m
            conf_mask[m > 0] = np.maximum(conf_mask[m > 0], confs[i])

        if open_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (open_k, open_k))
            union_mask = cv2.morphologyEx(union_mask, cv2.MORPH_OPEN, k)

        if close_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (close_k, close_k))
            union_mask = cv2.morphologyEx(union_mask, cv2.MORPH_CLOSE, k)

        if dilate_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (dilate_k, dilate_k))
            union_mask = cv2.dilate(union_mask, k, iterations=1)

        num_labels, cc_map, stats, _ = cv2.connectedComponentsWithStats(union_mask, connectivity=8)
        debug_stats[f"{cls_name}_cc_total"] += int(max(0, num_labels - 1))

        for cc_id in range(1, num_labels):
            area = int(stats[cc_id, cv2.CC_STAT_AREA])
            if area < min_area or area > img_area * 0.95:
                continue

            debug_stats[f"{cls_name}_cc_after_area"] += 1

            x, y, bw, bh = stats[cc_id, :4]
            x1 = float(max(0, x - box_pad))
            y1 = float(max(0, y - box_pad))
            x2 = float(min(w - 1, x + bw + box_pad))
            y2 = float(min(h - 1, y + bh + box_pad))

            box = expand_box([x1, y1, x2, y2], w, h, expand_ratio)
            region_conf = float(np.max(conf_mask[cc_map == cc_id]))
            region_area = box_area(box)

            proposals.append(
                {
                    "xyxy": box,
                    "conf": region_conf,
                    "cls_name": cls_name,
                    "area": region_area,
                    "rank_score": rank_score(cls_name, region_conf, region_area, img_area),
                }
            )

    for cls_name in ["fire", "smoke"]:
        debug_stats[f"{cls_name}_props_before_post"] = sum(
            1 for p in proposals if p.get("cls_name") == cls_name
        )

    return proposals, dict(debug_stats)


def overlay_simple(img_bgr: np.ndarray, props: List[Dict[str, Any]]) -> np.ndarray:
    canvas = img_bgr.copy()
    for p in props:
        if not isinstance(p, dict) or "xyxy" not in p:
            continue
        x1, y1, x2, y2 = map(int, p["xyxy"])
        cls_name = str(p.get("cls_name", "obj"))
        conf = float(p.get("conf", 0.0))
        color = (0, 0, 255) if cls_name == "fire" else (128, 128, 128)
        cv2.rectangle(canvas, (x1, y1), (x2, y2), color, 2)
        cv2.putText(
            canvas,
            f"{cls_name}:{conf:.2f}",
            (x1, max(20, y1 - 6)),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.6,
            color,
            2,
            cv2.LINE_AA,
        )
    return canvas


def infer_route_column(df: pd.DataFrame, route_column: str) -> str:
    if route_column != "auto":
        if route_column in df.columns:
            return route_column
        raise KeyError(f"manifest 中不存在 route 列: {route_column}")

    for col in ["stage1_route", "route"]:
        if col in df.columns:
            return col

    raise KeyError("manifest 中未找到 route/stage1_route 列。")


def find_image_by_key(images_dir: Path, key: str, valid_exts: set[str], recursive: bool) -> Optional[Path]:
    key = sanitize_key(key)

    for ext in valid_exts:
        candidate = images_dir / f"{key}{ext}"
        if candidate.exists():
            return candidate

    key_path = images_dir / key
    if key_path.exists() and key_path.suffix.lower() in valid_exts:
        return key_path

    target_stem = Path(key).stem
    iterator = images_dir.rglob("*") if recursive else images_dir.iterdir()
    for p in iterator:
        if p.is_file() and p.suffix.lower() in valid_exts and p.stem == target_stem:
            return p

    return None


def make_key_from_path(path: Path, root: Path) -> str:
    try:
        return str(path.relative_to(root).with_suffix("")).replace("\\", "/")
    except Exception:
        return path.stem


def resolve_selected_images(args: argparse.Namespace, images_dir: Path, valid_exts: set[str]):
    manifest_csv = Path(args.manifest_csv) if args.manifest_csv else None

    if manifest_csv is None:
        files = sorted(
            [
                p
                for p in (images_dir.rglob("*") if args.recursive_images else images_dir.iterdir())
                if p.is_file() and p.suffix.lower() in valid_exts
            ]
        )
        return [
            {
                "sample_key": make_key_from_path(p, images_dir),
                "image_path": p,
                "source_route": None,
                "next_route_if_sam_has_boxes": args.default_next_route_if_sam_has_boxes,
                "next_route_if_sam_empty": args.default_next_route_if_sam_empty,
                "manifest_row": {},
            }
            for p in files
        ], None, None

    df = pd.read_csv(manifest_csv)
    real_route_col = infer_route_column(df, args.route_column)
    df = df[df[real_route_col].astype(str).isin(args.include_routes)].copy()

    selected = []
    for _, row in df.iterrows():
        row_dict = row.to_dict()
        key = sanitize_key(safe_str(row_dict.get(args.key_column, "")))
        img_path: Optional[Path] = None

        if args.image_path_column in row_dict and safe_str(row_dict.get(args.image_path_column)):
            candidate = Path(safe_str(row_dict.get(args.image_path_column)))
            if candidate.exists():
                img_path = candidate

        if img_path is None and key:
            img_path = find_image_by_key(images_dir, key, valid_exts, recursive=args.recursive_images)

        if img_path is None:
            continue

        selected.append(
            {
                "sample_key": key if key else make_key_from_path(img_path, images_dir),
                "image_path": img_path,
                "source_route": safe_str(row_dict.get(real_route_col, "")),
                "next_route_if_sam_has_boxes": safe_str(
                    row_dict.get(args.next_route_if_sam_has-boxes-column, args.default_next_route_if_sam_has_boxes),
                    args.default_next_route_if_sam_has_boxes,
                ) if False else safe_str(  # placeholder avoided below
                    row_dict.get(args.next_route_if_sam_has_boxes_column, args.default_next_route_if_sam_has_boxes),
                    args.default_next_route_if_sam_has_boxes,
                ),
                "next_route_if_sam_empty": safe_str(
                    row_dict.get(args.next_route_if_sam_empty_column, args.default_next_route_if_sam_empty),
                    args.default_next_route_if_sam_empty,
                ),
                "manifest_row": row_dict,
            }
        )

    return selected, df, real_route_col

def prop_to_candidate_row(
    sample_key: str,
    image_path: Path,
    image_w: int,
    image_h: int,
    prop: Dict[str, Any],
    review_stage: str,
    crop_expand_ratio: float,
) -> Dict[str, Any]:
    cls_name = str(prop.get("cls_name", ""))
    cls_id = CLASS_TO_ID.get(cls_name, -1)
    x1, y1, x2, y2 = [float(v) for v in prop["xyxy"]]
    bw = max(0.0, x2 - x1)
    bh = max(0.0, y2 - y1)
    xc = (x1 + x2) / 2.0 / max(1.0, image_w)
    yc = (y1 + y2) / 2.0 / max(1.0, image_h)
    wn = bw / max(1.0, image_w)
    hn = bh / max(1.0, image_h)
    area_ratio = (bw * bh) / max(1.0, float(image_w * image_h))
    return {
        "sample_key": sanitize_key(sample_key),
        "image_path": str(image_path),
        "box_uid": make_box_uid(sample_key, "sam", cls_id, xc, yc, wn, hn),
        "source": "sam",
        "review_stage": review_stage,
        "cls_id": cls_id,
        "cls_name": cls_name,
        "x1": round(x1, 6),
        "y1": round(y1, 6),
        "x2": round(x2, 6),
        "y2": round(y2, 6),
        "xc": round(xc, 6),
        "yc": round(yc, 6),
        "w": round(wn, 6),
        "h": round(hn, 6),
        "sam_conf": round(float(prop.get("conf", 0.0)), 6),
        "box_area_ratio": round(area_ratio, 6),
        "touch_border": int(x1 <= 1 or y1 <= 1 or x2 >= image_w - 2 or y2 >= image_h - 2),
        "box_risk_score": 0.0,
        "risk_tags": "sam_candidate",
        "crop_expand_ratio": crop_expand_ratio,
    }


def build_predictor(model_path: Optional[str], conf: float, iou: float, imgsz: int, half: bool):
    if SAM3SemanticPredictor is None:
        raise ImportError("无法导入 ultralytics.models.sam.SAM3SemanticPredictor") from SAM_IMPORT_ERROR

    overrides: Dict[str, Any] = {
        "conf": conf,
        "iou": iou,
        "agnostic_nms": False,
        "task": "segment",
        "mode": "predict",
        "save": False,
        "half": bool(half),
        "imgsz": int(imgsz),
    }
    if model_path:
        overrides["model"] = model_path

    try:
        return SAM3SemanticPredictor(overrides=overrides)
    except TypeError:
        try:
            return SAM3SemanticPredictor(model=model_path, conf=conf, iou=iou, imgsz=imgsz, half=half)
        except TypeError:
            return SAM3SemanticPredictor(model_path)

def main() -> None:
    args = parse_args()

    images_dir = Path(args.images_dir)
    output_dir = Path(args.output_dir)
    labels_root = output_dir / "labels"
    report_dir = output_dir / "report"
    cache_dir = output_dir / "cache"
    vis_dir = output_dir / "vis"

    ensure_dir(labels_root)
    ensure_dir(report_dir)
    if args.disk_cache:
        ensure_dir(cache_dir)
    if args.save_vis:
        ensure_dir(vis_dir)

    valid_exts = {str(x).lower() for x in args.valid_exts}
    selected, _, _ = resolve_selected_images(args, images_dir, valid_exts)
    print(f"[INFO] selected {'/'.join(args.include_routes)} images: {len(selected)}")
    print(f"[INFO] disk_cache={bool(args.disk_cache)}, max_raw_masks_per_class={int(args.max_raw_masks_per_class)}")

    params = merge_params(DEFAULT_PARAMS, BASE_CONFIG if isinstance(BASE_CONFIG, dict) else {})
    model_path = args.model_path or (BASE_CONFIG.get("MODEL_PATH") if isinstance(BASE_CONFIG, dict) else None)
    prompts = args.prompts or (BASE_CONFIG.get("PROMPTS") if isinstance(BASE_CONFIG, dict) else None) or ["fire", "smoke"]

    predictor = build_predictor(
        model_path=model_path,
        conf=args.predictor_conf,
        iou=args.predictor_iou,
        imgsz=args.predictor_imgsz,
        half=args.half,
    )

    send_vlm_rows: List[Dict[str, Any]] = []
    sam_empty_rows: List[Dict[str, Any]] = []
    vlm_candidate_rows: List[Dict[str, Any]] = []

    for idx, sample in enumerate(selected, 1):
        sample_key = sanitize_key(sample["sample_key"])
        image_path = Path(sample["image_path"])
        print(f"[{idx}/{len(selected)}] {image_path.name}")

        img_bgr = cv2.imread(str(image_path))
        if img_bgr is None:
            print("  -> skip unreadable image")
            continue

        h, w = img_bgr.shape[:2]
        image_already_set = predictor_set_image(predictor, str(image_path))

        all_props: List[Dict[str, Any]] = []
        stage_stats_sum: Dict[str, int] = defaultdict(int)

        for prompt in prompts:
            npz_path = None
            if args.disk_cache:
                npz_path = cache_file_of(cache_dir, sample_key or image_path.stem, image_path.name, str(prompt))

            payload = build_payload_for_prompt(
                predictor=predictor,
                image_path=str(image_path),
                prompt=str(prompt),
                npz_path=npz_path,
                force_recache=args.force_recache,
                use_disk_cache=bool(args.disk_cache),
                max_raw_masks_per_class=int(args.max_raw_masks_per_class),
                image_already_set=image_already_set,
                image_hw=(h, w),
            )

            props, stats = build_region_proposals_from_payload(
                payload=payload,
                params=params,
                prompt_hint=str(prompt),
            )
            all_props.extend(props)

            for k, v in stats.items():
                stage_stats_sum[k] += int(v)

            try:
                del payload
            except Exception:
                pass
            maybe_empty_cache()

        if all_props:
            final_props, post_stats = postprocess_v2(all_props, params, w, h)
        else:
            final_props, post_stats = [], zero_post_stats()

        label_path = make_label_path(labels_root, sample_key, image_path)
        num_boxes = save_yolo_labels(label_path, final_props, w, h)

        common_row = dict(sample.get("manifest_row", {}))
        common_row.update(
            {
                "sample_key": sample_key,
                "image_path": str(image_path),
                "sam_label_path": str(label_path),
                "sam_num_boxes": safe_int(num_boxes, 0),
                "source_route": sample.get("source_route"),
            }
        )
        common_row.update(stage_stats_sum)
        common_row.update(post_stats)

        if num_boxes > 0:
            route_name = safe_str(sample.get("next_route_if_sam_has_boxes"), args.default_next_route_if_sam_has_boxes)
            common_row["next_route"] = route_name
            if args.copy_vlm_images:
                common_row["copied_image_path"] = copy_image_for_vlm(
                    image_path,
                    output_dir,
                    route_name,
                    sample_key or image_path.stem,
                )
            send_vlm_rows.append(common_row)

            if args.export_vlm_after_sam_candidates:
                for prop in final_props:
                    if not isinstance(prop, dict) or "xyxy" not in prop:
                        continue
                    vlm_candidate_rows.append(
                        prop_to_candidate_row(
                            sample_key=sample_key or image_path.stem,
                            image_path=image_path,
                            image_w=w,
                            image_h=h,
                            prop=prop,
                            review_stage="send_vlm_after_sam",
                            crop_expand_ratio=args.vlm_crop_expand_ratio,
                        )
                    )
        else:
            route_name = safe_str(sample.get("next_route_if_sam_empty"), args.default_next_route_if_sam_empty)
            common_row["next_route"] = route_name
            sam_empty_rows.append(common_row)

        if args.save_vis:
            vis_path = (vis_dir / (sample_key or image_path.stem)).with_suffix(
                image_path.suffix if image_path.suffix else ".jpg"
            )
            ensure_dir(vis_path.parent)
            cv2.imwrite(str(vis_path), overlay_simple(img_bgr, final_props))

        predictor_reset_image(predictor)
        maybe_empty_cache()

    write_csv(report_dir / "send_vlm_after_sam_manifest.csv", send_vlm_rows)
    write_csv(report_dir / "sam_empty_manifest.csv", sam_empty_rows)

    if args.export_vlm_after_sam_candidates:
        write_csv(report_dir / "vlm_after_sam_candidates.csv", vlm_candidate_rows)

    summary = {
        "selected_images": len(selected),
        "send_vlm_after_sam": len(send_vlm_rows),
        "sam_empty": len(sam_empty_rows),
        "vlm_candidates": len(vlm_candidate_rows),
        "labels_root": str(labels_root),
        "report_dir": str(report_dir),
        "cache_dir": str(cache_dir),
        "disk_cache": bool(args.disk_cache),
        "predictor_half": bool(args.half),
        "predictor_imgsz": int(args.predictor_imgsz),
        "max_raw_masks_per_class": int(args.max_raw_masks_per_class),
        "model_path": model_path,
    }

    with open(report_dir / "summary.json", "w", encoding="utf-8") as f:
        json.dump(summary, f, ensure_ascii=False, indent=2)

    print("[DONE] sam selective generation finished")
    print(json.dumps(summary, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()

        "fire_raw_masks": 0,
        "fire_after_conf": 0,
        "fire_cc_total": 0,
        "fire_cc_after_area": 0,
        "fire_props_before_post": 0,

        "smoke_raw_masks": 0,
        "smoke_after_conf": 0,
        "smoke_cc_total": 0,
        "smoke_cc_after_area": 0,
        "smoke_props_before_post": 0,
    }

    masks = payload["masks"]
    confs = payload["confs"]
    clss = payload["clss"]
    names = payload["names"]
    H = payload["H"]
    W = payload["W"]

    if len(confs) == 0 or masks.size == 0:
        return [], debug_stats

    img_area = H * W
    proposals = []

    unique_clss = np.unique(clss)
    for cls_id in unique_clss:
        cls_name = names[int(cls_id)]
        if cls_name not in ["fire", "smoke"]:
            continue

        conf_thr = params[f"{cls_name}_conf"]
        min_area = params[f"{cls_name}_min_area"]
        open_k = params[f"{cls_name}_open"]
        close_k = params[f"{cls_name}_close"]
        dilate_k = params[f"{cls_name}_dilate"]
        box_pad = params[f"{cls_name}_box_pad"]
        expand_ratio = params[f"{cls_name}_expand_ratio"]

        raw_idx = np.where(clss == cls_id)[0]
        debug_stats[f"{cls_name}_raw_masks"] += int(len(raw_idx))

        idx = np.where((clss == cls_id) & (confs >= conf_thr))[0]
        debug_stats[f"{cls_name}_after_conf"] += int(len(idx))
        if len(idx) == 0:
            continue

        union_mask = np.zeros((H, W), dtype=np.uint8)
        conf_mask = np.zeros((H, W), dtype=np.float32)

        for i in idx:
            m = masks[i]
            union_mask |= m
            conf_mask[m > 0] = np.maximum(conf_mask[m > 0], confs[i])

        if open_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (open_k, open_k))
            union_mask = cv2.morphologyEx(union_mask, cv2.MORPH_OPEN, k)

        if close_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (close_k, close_k))
            union_mask = cv2.morphologyEx(union_mask, cv2.MORPH_CLOSE, k)

        if dilate_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (dilate_k, dilate_k))
            union_mask = cv2.dilate(union_mask, k, iterations=1)

        num_labels, cc_map, stats, _ = cv2.connectedComponentsWithStats(union_mask, connectivity=8)
        debug_stats[f"{cls_name}_cc_total"] += int(max(0, num_labels - 1))

        for cc_id in range(1, num_labels):
            area = int(stats[cc_id, cv2.CC_STAT_AREA])

            if area < min_area:
                continue
            if area > img_area * 0.95:
                continue

            debug_stats[f"{cls_name}_cc_after_area"] += 1

            x, y, w, h = stats[cc_id, :4]
            x1 = float(max(0, x - box_pad))
            y1 = float(max(0, y - box_pad))
            x2 = float(min(W - 1, x + w + box_pad))
            y2 = float(min(H - 1, y + h + box_pad))

            box = expand_box([x1, y1, x2, y2], W, H, expand_ratio)
            region_conf = float(np.max(conf_mask[cc_map == cc_id]))
            region_area = box_area(box)

            proposals.append({
                "xyxy": box,
                "conf": region_conf,
                "cls_name": cls_name,
                "area": region_area,
                "rank_score": rank_score(cls_name, region_conf, region_area, img_area),
            })

    for cls_name in ["fire", "smoke"]:
        debug_stats[f"{cls_name}_props_before_post"] = sum(
            1 for p in proposals if p["cls_name"] == cls_name
        )

    return proposals, debug_stats


def build_or_load_cache(base_cfg, cache_dir, force_recache=False):
    image_dir = Path(base_cfg["IMAGE_DIR"])
    label_dir = Path(base_cfg["LABEL_DIR"])
    prompts = base_cfg["PROMPTS"]

    image_files = sorted([
        p for p in image_dir.iterdir()
        if p.suffix.lower() in base_cfg["VALID_EXTS"]
    ])

    # 用尽量宽松的 predictor，后面 trial 再做阈值筛选
    overrides = dict(
        conf=0.01,
        iou=0.95,
        agnostic_nms=False,
        task="segment",
        mode="predict",
        model=base_cfg["MODEL_PATH"],
        half=True,
        save=False
    )
    predictor = SAM3SemanticPredictor(overrides=overrides)

    dataset = []

    for idx, img_path in enumerate(image_files, 1):
        print(f"[CACHE {idx}/{len(image_files)}] {img_path.name}")

        img_bgr = cv2.imread(str(img_path))
        if img_bgr is None:
            print("  -> skip unreadable image")
            continue

        H, W = img_bgr.shape[:2]
        label_path = label_dir / f"{img_path.stem}.txt"
        gts = read_yolo_label(label_path, W, H)

        predictor.set_image(str(img_path))

        prompt_cache_files = {}
        for prompt in prompts:
            npz_path = cache_file_of(cache_dir, img_path.name, prompt)
            prompt_cache_files[prompt] = npz_path

            if npz_path.exists() and not force_recache:
                continue

            try:
                results = predictor(text=[prompt])
            except Exception as e:
                print(f"  -> predictor error on prompt={prompt}: {e}")
                save_payload(
                    npz_path,
                    np.empty((0, H, W), dtype=np.uint8),
                    np.empty((0,), dtype=np.float32),
                    np.empty((0,), dtype=np.int32),
                    {},
                    H, W
                )
                continue

            masks_list = []
            confs_list = []
            clss_list = []
            names = {}

            for result in results:
                if result.boxes is None or len(result.boxes) == 0 or result.masks is None:
                    continue

                masks = result.masks.data.cpu().numpy().astype(np.uint8)
                confs = result.boxes.conf.cpu().numpy().astype(np.float32)
                clss = result.boxes.cls.cpu().numpy().astype(np.int32)
                names = result.names

                if len(confs) > 0:
                    masks_list.append(masks)
                    confs_list.append(confs)
                    clss_list.append(clss)

            if len(confs_list) == 0:
                save_payload(
                    npz_path,
                    np.empty((0, H, W), dtype=np.uint8),
                    np.empty((0,), dtype=np.float32),
                    np.empty((0,), dtype=np.int32),
                    names,
                    H, W
                )
            else:
                save_payload(
                    npz_path,
                    np.concatenate(masks_list, axis=0),
                    np.concatenate(confs_list, axis=0),
                    np.concatenate(clss_list, axis=0),
                    names,
                    H, W
                )

        dataset.append({
            "image_name": img_path.name,
            "image_path": str(img_path),
            "H": H,
            "W": W,
            "gts": gts,
            "prompt_cache_files": {k: str(v) for k, v in prompt_cache_files.items()},
        })

    return dataset


def make_trial_params_list(base_params, search_space, max_trials, random_seed):
    rng = random.Random(random_seed)
    trials = []
    seen = set()

    def add_params(params):
        sig = params_signature(params)
        if sig not in seen:
            seen.add(sig)
            trials.append(params)

    # 先加 base
    add_params(deepcopy(base_params))

    # 再加 manual anchors
    for override in MANUAL_TRIAL_OVERRIDES:
        params = deepcopy(base_params)
        params.update(override)
        add_params(params)

    # 剩下随机采样
    while len(trials) < max_trials:
        params = deepcopy(base_params)
        for k, choices in search_space.items():
            params[k] = rng.choice(choices)
        add_params(params)

    return trials[:max_trials]


def evaluate_trial(dataset, base_cfg, trial_params, save_dir=None, save_vis=False):
    rows = []
    totals = defaultdict(float)

    vis_dir = None
    if save_dir is not None:
        ensure_dir(save_dir)
        vis_dir = Path(save_dir) / "vis"
        if save_vis:
            ensure_dir(vis_dir)

    for sample in dataset:
        image_name = sample["image_name"]
        img_w = sample["W"]
        img_h = sample["H"]
        gts = sample["gts"]

        all_proposals = []
        stage_stats_sum = defaultdict(int)

        for prompt in base_cfg["PROMPTS"]:
            payload = load_payload(sample["prompt_cache_files"][prompt])
            proposals, stats = build_region_proposals_from_payload(payload, trial_params)
            all_proposals.extend(proposals)
            for k, v in stats.items():
                stage_stats_sum[k] += v

        final_props, post_stats = postprocess_v2(all_proposals, trial_params, img_w, img_h)

        fire_preds = [p for p in final_props if p["cls_name"] == "fire"]
        smoke_preds = [p for p in final_props if p["cls_name"] == "smoke"]
        fire_gts = [g for g in gts if g["cls_name"] == "fire"]
        smoke_gts = [g for g in gts if g["cls_name"] == "smoke"]

        fire_tp, fire_fp, fire_fn, fire_unmatched_preds, fire_unmatched_gts = greedy_match(
            fire_preds, fire_gts, base_cfg["MATCH_IOU_THR"]
        )
        smoke_tp, smoke_fp, smoke_fn, smoke_unmatched_preds, smoke_unmatched_gts = greedy_match(
            smoke_preds, smoke_gts, base_cfg["MATCH_IOU_THR"]
        )

        fire_precision = fire_tp / max(1, fire_tp + fire_fp)
        fire_recall = fire_tp / max(1, fire_tp + fire_fn)
        fire_f1 = 2 * fire_precision * fire_recall / max(1e-6, fire_precision + fire_recall)

        smoke_precision = smoke_tp / max(1, smoke_tp + smoke_fp)
        smoke_recall = smoke_tp / max(1, smoke_tp + smoke_fn)
        smoke_f1 = 2 * smoke_precision * smoke_recall / max(1e-6, smoke_precision + smoke_recall)

        row = {
            "image_name": image_name,

            "gt_fire": len(fire_gts),
            "gt_smoke": len(smoke_gts),
            "pred_fire": len(fire_preds),
            "pred_smoke": len(smoke_preds),

            "fire_tp": fire_tp,
            "fire_fp": fire_fp,
            "fire_fn": fire_fn,
            "fire_precision": fire_precision,
            "fire_recall": fire_recall,
            "fire_f1": fire_f1,

            "smoke_tp": smoke_tp,
            "smoke_fp": smoke_fp,
            "smoke_fn": smoke_fn,
            "smoke_precision": smoke_precision,
            "smoke_recall": smoke_recall,
            "smoke_f1": smoke_f1,
        }

        row.update(stage_stats_sum)
        row.update(post_stats)
        rows.append(row)

        for k, v in row.items():
            if k == "image_name":
                continue
            if isinstance(v, (int, float)):
                totals[k] += v

        if save_vis:
            img_bgr = cv2.imread(sample["image_path"])
            unmatched_preds_all = fire_unmatched_preds + smoke_unmatched_preds
            unmatched_gts_all = fire_unmatched_gts + smoke_unmatched_gts
            draw_diagnosis(
                img_bgr,
                final_props,
                gts,
                unmatched_preds_all,
                unmatched_gts_all,
                vis_dir / image_name
            )

    fire_precision = totals["fire_tp"] / max(1, totals["fire_tp"] + totals["fire_fp"])
    fire_recall = totals["fire_tp"] / max(1, totals["fire_tp"] + totals["fire_fn"])
    fire_f1 = 2 * fire_precision * fire_recall / max(1e-6, fire_precision + fire_recall)

    smoke_precision = totals["smoke_tp"] / max(1, totals["smoke_tp"] + totals["smoke_fp"])
    smoke_recall = totals["smoke_tp"] / max(1, totals["smoke_tp"] + totals["smoke_fn"])
    smoke_f1 = 2 * smoke_precision * smoke_recall / max(1e-6, smoke_precision + smoke_recall)

    summary = {
        "num_images": len(rows),
        "params": trial_params,
        "totals": dict(totals),
        "final_metrics": {
            "fire_precision": fire_precision,
            "fire_recall": fire_recall,
            "fire_f1": fire_f1,
            "smoke_precision": smoke_precision,
            "smoke_recall": smoke_recall,
            "smoke_f1": smoke_f1
        }
    }

    score = score_from_metrics(fire_f1, smoke_f1)

    if save_dir is not None:
        csv_path = Path(save_dir) / "per_image_diagnosis.csv"
        summary_path = Path(save_dir) / "summary.json"

        if rows:
            fieldnames = list(rows[0].keys())
            with open(csv_path, "w", encoding="utf-8", newline="") as f:
                writer = csv.DictWriter(f, fieldnames=fieldnames)
                writer.writeheader()
                writer.writerows(rows)

        with open(summary_path, "w", encoding="utf-8") as f:
            json.dump(summary, f, ensure_ascii=False, indent=2)

    return score, summary, rows

            f"fire_f1={item['fire_f1']:.6f} | smoke_f1={item['smoke_f1']:.6f}"

"""
作用就是：让 AI 去跑一遍测试集，然后自动和人工标注的“标准答案”对答案，最后算出这行 AI 到底准不准，并生成详细的成绩单。

核心循环（逐张图片评估）
程序会对每一张图片执行以下动作：
读取数据：加载图片本身，并读取对应的 YOLO 格式标签文件（即“标准答案”）。
模型推理：
给 AI 输入提示词（Prompt），比如 "fire" 或 "smoke"。
AI 返回初步的分割结果。
结果精炼（后处理）：
build_region_proposals_v2 和 postprocess_v2：这两个函数负责把 AI 原始的分割数据变成一个一个的“检测框”，并过滤掉那些太小的、重叠的或者是置信度太低的“垃圾结果”。

3. “对答案”逻辑（贪婪匹配）
这是评估中最关键的一步。程序把 AI 预测出的框（Preds）和人工标注的框（GTs）放在一起比对：
greedy_match（贪婪匹配）：计算两个框的交并比（IoU）。
TP (True Positive)：AI 猜对了。预测框和答案框重合度高。
FP (False Positive)：AI 报假警。AI 说这儿有火，其实没火（误报）。
FN (False Negative)：AI 漏检了。这儿明明有火，AI 没看见（漏报）。
4. 计算指标（衡量模型好坏）
针对“火”和“烟”分别计算三个核心指标：
Precision (精确率)：你报的警里面，有多少是真的？（反映误报情况）
Recall (召回率)：所有的火灾里面，你抓住了多少？（反映漏报情况）
F1-score：一个综合精确率和召回率的平均分，分数越高说明模型综合实力越强。
5. 保存结果（输出报告）
画图可视化 (draw_diagnosis)：在原图上把 AI 猜的框和真实的框都画出来，并标记出哪里猜对了，哪里漏了，哪里多报了。这是最直观的差错分析工具。
生成 CSV 报表：把每一张图片的得分（TP/FP/FN）存入 Excel 格式的表里，方便后期数据分析。
生成汇总 JSON：最后给出一个总分，告诉你这次测试的整体表现如何。

def ensure_dir(p):
    Path(p).mkdir(parents=True, exist_ok=True)


def xywhn_to_xyxy(line, img_w, img_h):
    parts = line.strip().split()
    if len(parts) != 5:
        return None
    cls_id = int(parts[0])
    cx, cy, w, h = map(float, parts[1:])
    x1 = (cx - w / 2.0) * img_w
    y1 = (cy - h / 2.0) * img_h
    x2 = (cx + w / 2.0) * img_w
    y2 = (cy + h / 2.0) * img_h
    return cls_id, [x1, y1, x2, y2]


def read_yolo_label(label_path, img_w, img_h):
    gts = []
    if not Path(label_path).exists():
        return gts

    with open(label_path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parsed = xywhn_to_xyxy(line, img_w, img_h)
            if parsed is None:
                continue
            cls_id, box = parsed
            cls_name = CONFIG["CLASS_MAP"].get(cls_id, None)
            if cls_name is None:
                continue
            gts.append({
                "cls_name": cls_name,
                "xyxy": box
            })
    return gts


def box_area(box):
    return max(0.0, box[2] - box[0]) * max(0.0, box[3] - box[1])


def box_iou(box_a, box_b):
    inter_x1 = max(box_a[0], box_b[0])
    inter_y1 = max(box_a[1], box_b[1])
    inter_x2 = min(box_a[2], box_b[2])
    inter_y2 = min(box_a[3], box_b[3])

    inter_w = max(0.0, inter_x2 - inter_x1)
    inter_h = max(0.0, inter_y2 - inter_y1)
    inter = inter_w * inter_h

    area_a = max(1e-6, box_area(box_a))
    area_b = max(1e-6, box_area(box_b))
    union = area_a + area_b - inter + 1e-6
    return inter / union


def box_iof(box_a, box_b):
    # box_b 有多少面积被 box_a 包住
    inter_x1 = max(box_a[0], box_b[0])
    inter_y1 = max(box_a[1], box_b[1])
    inter_x2 = min(box_a[2], box_b[2])
    inter_y2 = min(box_a[3], box_b[3])

    inter_w = max(0.0, inter_x2 - inter_x1)
    inter_h = max(0.0, inter_y2 - inter_y1)
    inter_area = inter_w * inter_h

    area_b = max(1e-6, box_area(box_b))
    return inter_area / area_b


def nms_numpy(boxes, scores, iou_thr=0.5):
    if len(boxes) == 0:
        return []

    order = scores.argsort()[::-1]
    keep = []

    while len(order) > 0:
        i = order[0]
        keep.append(i)
        if len(order) == 1:
            break

        rest = order[1:]

        xx1 = np.maximum(boxes[i, 0], boxes[rest, 0])
        yy1 = np.maximum(boxes[i, 1], boxes[rest, 1])
        xx2 = np.minimum(boxes[i, 2], boxes[rest, 2])
        yy2 = np.minimum(boxes[i, 3], boxes[rest, 3])

        inter_w = np.maximum(0.0, xx2 - xx1)
        inter_h = np.maximum(0.0, yy2 - yy1)
        inter = inter_w * inter_h

        area_i = np.maximum(1e-6, (boxes[i, 2] - boxes[i, 0]) * (boxes[i, 3] - boxes[i, 1]))
        area_r = np.maximum(1e-6, (boxes[rest, 2] - boxes[rest, 0]) * (boxes[rest, 3] - boxes[rest, 1]))
        union = area_i + area_r - inter + 1e-6
        iou = inter / union

        order = rest[iou <= iou_thr]

    return keep


def expand_box(box, img_w, img_h, ratio=0.0):
    if ratio <= 0:
        return box

    x1, y1, x2, y2 = box
    w = x2 - x1
    h = y2 - y1
    dw = w * ratio / 2.0
    dh = h * ratio / 2.0

    nx1 = max(0.0, x1 - dw)
    ny1 = max(0.0, y1 - dh)
    nx2 = min(float(img_w - 1), x2 + dw)
    ny2 = min(float(img_h - 1), y2 + dh)
    return [nx1, ny1, nx2, ny2]


def box_gap(box_a, box_b):
    ax1, ay1, ax2, ay2 = box_a
    bx1, by1, bx2, by2 = box_b
    gap_x = max(0.0, max(ax1, bx1) - min(ax2, bx2))
    gap_y = max(0.0, max(ay1, by1) - min(ay2, by2))
    return gap_x, gap_y


def rank_score(cls_name, conf, area, img_area):
    area_ratio = area / max(1.0, img_area)

    if cls_name == "fire":
        # 给 fire 的面积一些加成，避免只保留很小但高置信的小火芯
        area_bonus = min(0.25, area_ratio / 0.01 * 0.25)
        return float(conf + area_bonus)

    # smoke 保守一些，主要看 conf
    area_bonus = min(0.10, area_ratio / 0.04 * 0.10)
    return float(conf + area_bonus)


def merge_boxes_same_class(proposals, cls_name, params, img_w, img_h):
    if not proposals:
        return []

    merge_iou = params[f"{cls_name}_merge_iou"]
    merge_gap = params[f"{cls_name}_merge_gap"]

    proposals = sorted(proposals, key=lambda x: x["rank_score"], reverse=True)
    merged = []

    while proposals:
        current = proposals.pop(0)
        cx1, cy1, cx2, cy2 = current["xyxy"]

        changed = True
        while changed:
            changed = False
            remain = []

            for p in proposals:
                iou = box_iou([cx1, cy1, cx2, cy2], p["xyxy"])
                gx, gy = box_gap([cx1, cy1, cx2, cy2], p["xyxy"])
                is_close = (gx <= merge_gap and gy <= merge_gap)

                if iou >= merge_iou or is_close:
                    px1, py1, px2, py2 = p["xyxy"]
                    cx1 = min(cx1, px1)
                    cy1 = min(cy1, py1)
                    cx2 = max(cx2, px2)
                    cy2 = max(cy2, py2)

                    current["conf"] = max(current["conf"], p["conf"])
                    current["xyxy"] = [cx1, cy1, cx2, cy2]
                    current["area"] = box_area(current["xyxy"])
                    current["rank_score"] = rank_score(
                        cls_name, current["conf"], current["area"], img_w * img_h
                    )
                    changed = True
                else:
                    remain.append(p)

            proposals = remain

        merged.append(current)

    return merged


def remove_contained_boxes(proposals, iof_thr):
    if not proposals:
        return []

    proposals = sorted(proposals, key=lambda x: x["area"], reverse=True)
    kept = []

    for p in proposals:
        drop = False
        for q in kept:
            if box_iof(q["xyxy"], p["xyxy"]) > iof_thr:
                drop = True
                break
        if not drop:
            kept.append(p)

    return kept


def greedy_match(preds, gts, iou_thr=0.5):
    if len(preds) == 0 and len(gts) == 0:
        return 0, 0, 0, [], []

    matched_gt = set()
    matched_pred = set()

    pairs = []
    for pi, p in enumerate(preds):
        for gi, g in enumerate(gts):
            iou = box_iou(p["xyxy"], g["xyxy"])
            pairs.append((iou, pi, gi))

    pairs.sort(reverse=True, key=lambda x: x[0])

    for iou, pi, gi in pairs:
        if iou < iou_thr:
            break
        if pi in matched_pred or gi in matched_gt:
            continue
        matched_pred.add(pi)
        matched_gt.add(gi)

    tp = len(matched_pred)
    fp = len(preds) - tp
    fn = len(gts) - tp

    unmatched_preds = [preds[i] for i in range(len(preds)) if i not in matched_pred]
    unmatched_gts = [gts[i] for i in range(len(gts)) if i not in matched_gt]

    return tp, fp, fn, unmatched_preds, unmatched_gts


def build_region_proposals_v2(result, params):
    debug_stats = {
        "fire_raw_masks": 0,
        "fire_after_conf": 0,
        "fire_cc_total": 0,
        "fire_cc_after_area": 0,
        "fire_props_before_post": 0,

        "smoke_raw_masks": 0,
        "smoke_after_conf": 0,
        "smoke_cc_total": 0,
        "smoke_cc_after_area": 0,
        "smoke_props_before_post": 0,
    }

    if result.masks is None or result.boxes is None or len(result.boxes) == 0:
        return [], debug_stats

    masks = result.masks.data.cpu().numpy().astype(np.uint8)
    confs = result.boxes.conf.cpu().numpy()
    clss = result.boxes.cls.cpu().numpy().astype(int)
    names = result.names

    H, W = masks.shape[1:]
    img_area = H * W
    proposals = []

    unique_clss = np.unique(clss)
    for cls_id in unique_clss:
        cls_name = names[int(cls_id)]
        if cls_name not in ["fire", "smoke"]:
            continue

        conf_thr = params[f"{cls_name}_conf"]
        min_area = params[f"{cls_name}_min_area"]
        open_k = params[f"{cls_name}_open"]
        close_k = params[f"{cls_name}_close"]
        dilate_k = params[f"{cls_name}_dilate"]
        box_pad = params[f"{cls_name}_box_pad"]
        expand_ratio = params[f"{cls_name}_expand_ratio"]

        raw_idx = np.where(clss == cls_id)[0]
        debug_stats[f"{cls_name}_raw_masks"] += int(len(raw_idx))

        idx = np.where((clss == cls_id) & (confs >= conf_thr))[0]
        debug_stats[f"{cls_name}_after_conf"] += int(len(idx))
        if len(idx) == 0:
            continue

        union_mask = np.zeros((H, W), dtype=np.uint8)
        conf_mask = np.zeros((H, W), dtype=np.float32)

        for i in idx:
            m = masks[i]
            union_mask |= m
            conf_mask[m > 0] = np.maximum(conf_mask[m > 0], confs[i])

        if open_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (open_k, open_k))
            union_mask = cv2.morphologyEx(union_mask, cv2.MORPH_OPEN, k)

        if close_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (close_k, close_k))
            union_mask = cv2.morphologyEx(union_mask, cv2.MORPH_CLOSE, k)

        if dilate_k > 1:
            k = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (dilate_k, dilate_k))
            union_mask = cv2.dilate(union_mask, k, iterations=1)

        num_labels, cc_map, stats, _ = cv2.connectedComponentsWithStats(union_mask, connectivity=8)
        debug_stats[f"{cls_name}_cc_total"] += int(max(0, num_labels - 1))

        for cc_id in range(1, num_labels):
            area = int(stats[cc_id, cv2.CC_STAT_AREA])

            if area < min_area:
                continue
            if area > img_area * 0.95:
                continue

            debug_stats[f"{cls_name}_cc_after_area"] += 1

            x, y, w, h = stats[cc_id, :4]
            x1 = float(max(0, x - box_pad))
            y1 = float(max(0, y - box_pad))
            x2 = float(min(W - 1, x + w + box_pad))
            y2 = float(min(H - 1, y + h + box_pad))

            box = expand_box([x1, y1, x2, y2], W, H, expand_ratio)
            region_conf = float(np.max(conf_mask[cc_map == cc_id]))
            region_area = box_area(box)

            proposals.append({
                "xyxy": box,
                "conf": region_conf,
                "cls_name": cls_name,
                "area": region_area,
                "rank_score": rank_score(cls_name, region_conf, region_area, img_area)
            })

    for cls_name in ["fire", "smoke"]:
        debug_stats[f"{cls_name}_props_before_post"] = sum(
            1 for p in proposals if p["cls_name"] == cls_name
        )

    return proposals, debug_stats


def postprocess_v2(proposals, params, img_w, img_h):
    post_stats = {
        "fire_before_merge": 0,
        "fire_after_merge": 0,
        "fire_after_nms": 0,
        "fire_after_iof": 0,
        "fire_after_topk": 0,

        "smoke_before_merge": 0,
        "smoke_after_merge": 0,
        "smoke_after_nms": 0,
        "smoke_after_iof": 0,
        "smoke_after_topk": 0,
    }

    final_props = []

    for cls_name in ["fire", "smoke"]:
        cls_props = [p for p in proposals if p["cls_name"] == cls_name]
        post_stats[f"{cls_name}_before_merge"] = len(cls_props)

        if not cls_props:
            continue

        # fire 做更积极的 bbox merge；smoke 保守一些
        merged = merge_boxes_same_class(cls_props, cls_name, params, img_w, img_h)
        post_stats[f"{cls_name}_after_merge"] = len(merged)

        boxes = np.array([p["xyxy"] for p in merged], dtype=np.float32)
        scores = np.array([p["rank_score"] for p in merged], dtype=np.float32)

        keep_idx = nms_numpy(boxes, scores, iou_thr=params[f"{cls_name}_nms_iou"])
        after_nms = [merged[i] for i in keep_idx]
        post_stats[f"{cls_name}_after_nms"] = len(after_nms)

        iof_thr = params[f"{cls_name}_iof_thr"]
        after_iof = remove_contained_boxes(after_nms, iof_thr)
        post_stats[f"{cls_name}_after_iof"] = len(after_iof)

        # topk 用 rank_score 排，不再只按面积
        after_iof = sorted(after_iof, key=lambda x: x["rank_score"], reverse=True)
        topk = params[f"{cls_name}_topk"]
        after_topk = after_iof[:topk]
        post_stats[f"{cls_name}_after_topk"] = len(after_topk)

        final_props.extend(after_topk)

    return final_props, post_stats


def draw_diagnosis(image_bgr, preds, gts, unmatched_preds, unmatched_gts, save_path):
    out = image_bgr.copy()

    # GT: 绿色
    for g in gts:
        x1, y1, x2, y2 = map(int, g["xyxy"])
        color = (0, 255, 0)
        cv2.rectangle(out, (x1, y1), (x2, y2), color, 2)
        cv2.putText(out, f'GT-{g["cls_name"]}', (x1, max(20, y1 - 6)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.55, color, 2)

    # Pred: fire 红 / smoke 蓝
    for p in preds:
        x1, y1, x2, y2 = map(int, p["xyxy"])
        color = (255, 0, 0) if p["cls_name"] == "smoke" else (0, 0, 255)
        cv2.rectangle(out, (x1, y1), (x2, y2), color, 2)
        cv2.putText(out, f'P-{p["cls_name"]}', (x1, max(20, y2 + 18)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.55, color, 2)

    # FN: 黄色
    for g in unmatched_gts:
        x1, y1, x2, y2 = map(int, g["xyxy"])
        cv2.rectangle(out, (x1, y1), (x2, y2), (0, 255, 255), 3)
        cv2.putText(out, f'FN-{g["cls_name"]}', (x1, max(20, y1 - 24)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 255), 2)

    # FP: 品红
    for p in unmatched_preds:
        x1, y1, x2, y2 = map(int, p["xyxy"])
        cv2.rectangle(out, (x1, y1), (x2, y2), (255, 0, 255), 3)
        cv2.putText(out, f'FP-{p["cls_name"]}', (x1, max(20, y1 - 24)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 0, 255), 2)

    cv2.imwrite(str(save_path), out)


def main():
    output_dir = Path(CONFIG["OUTPUT_DIR"])
    vis_dir = output_dir / "vis"
    ensure_dir(output_dir)
    ensure_dir(vis_dir)

    csv_path = output_dir / "per_image_diagnosis.csv"
    summary_path = output_dir / "summary.json"

    params = CONFIG["PARAMS"]

    overrides = dict(
        conf=min(params["fire_conf"], params["smoke_conf"]),
        iou=max(params["fire_nms_iou"], params["smoke_nms_iou"]),
        agnostic_nms=False,
        task="segment",
        mode="predict",
        model=CONFIG["MODEL_PATH"],
        half=False,
        save=False
    )
    predictor = SAM3SemanticPredictor(overrides=overrides)

    image_dir = Path(CONFIG["IMAGE_DIR"])
    label_dir = Path(CONFIG["LABEL_DIR"])

    image_files = sorted([
        p for p in image_dir.iterdir()
        if p.suffix.lower() in CONFIG["VALID_EXTS"]
    ])

    rows = []
    totals = defaultdict(float)

    for idx, img_path in enumerate(image_files, 1):
        print(f"[{idx}/{len(image_files)}] {img_path.name}")

        img_bgr = cv2.imread(str(img_path))
        if img_bgr is None:
            print("  -> skip unreadable image")
            continue

        img_h, img_w = img_bgr.shape[:2]
        label_path = label_dir / f"{img_path.stem}.txt"
        gts = read_yolo_label(label_path, img_w, img_h)

        predictor.set_image(str(img_path))

        all_proposals = []
        stage_stats_sum = defaultdict(int)

        for prompt in CONFIG["PROMPTS"]:
            try:
                results = predictor(text=[prompt])
            except Exception as e:
                print(f"  -> predictor error on prompt={prompt}: {e}")
                continue

            for result in results:
                proposals, stats = build_region_proposals_v2(result, params)
                all_proposals.extend(proposals)
                for k, v in stats.items():
                    stage_stats_sum[k] += v

        final_props, post_stats = postprocess_v2(all_proposals, params, img_w, img_h)

        fire_preds = [p for p in final_props if p["cls_name"] == "fire"]
        smoke_preds = [p for p in final_props if p["cls_name"] == "smoke"]
        fire_gts = [g for g in gts if g["cls_name"] == "fire"]
        smoke_gts = [g for g in gts if g["cls_name"] == "smoke"]

        fire_tp, fire_fp, fire_fn, fire_unmatched_preds, fire_unmatched_gts = greedy_match(
            fire_preds, fire_gts, CONFIG["MATCH_IOU_THR"]
        )
        smoke_tp, smoke_fp, smoke_fn, smoke_unmatched_preds, smoke_unmatched_gts = greedy_match(
            smoke_preds, smoke_gts, CONFIG["MATCH_IOU_THR"]
        )

        fire_precision = fire_tp / max(1, fire_tp + fire_fp)
        fire_recall = fire_tp / max(1, fire_tp + fire_fn)
        fire_f1 = 2 * fire_precision * fire_recall / max(1e-6, fire_precision + fire_recall)

        smoke_precision = smoke_tp / max(1, smoke_tp + smoke_fp)
        smoke_recall = smoke_tp / max(1, smoke_tp + smoke_fn)
        smoke_f1 = 2 * smoke_precision * smoke_recall / max(1e-6, smoke_precision + smoke_recall)

        row = {
            "image_name": img_path.name,

            "gt_fire": len(fire_gts),
            "gt_smoke": len(smoke_gts),
            "pred_fire": len(fire_preds),
            "pred_smoke": len(smoke_preds),

            "fire_tp": fire_tp,
            "fire_fp": fire_fp,
            "fire_fn": fire_fn,
            "fire_precision": fire_precision,
            "fire_recall": fire_recall,
            "fire_f1": fire_f1,

            "smoke_tp": smoke_tp,
            "smoke_fp": smoke_fp,
            "smoke_fn": smoke_fn,
            "smoke_precision": smoke_precision,
            "smoke_recall": smoke_recall,
            "smoke_f1": smoke_f1,
        }

        row.update(stage_stats_sum)
        row.update(post_stats)
        rows.append(row)

        for k, v in row.items():
            if k == "image_name":
                continue
            if isinstance(v, (int, float)):
                totals[k] += v

        unmatched_preds_all = fire_unmatched_preds + smoke_unmatched_preds
        unmatched_gts_all = fire_unmatched_gts + smoke_unmatched_gts
        save_vis_path = vis_dir / img_path.name
        draw_diagnosis(
            img_bgr,
            final_props,
            gts,
            unmatched_preds_all,
            unmatched_gts_all,
            save_vis_path
        )

@dataclass
class Box:
    cls_id: int
    xc: float
    yc: float
    w: float
    h: float
    source: str = ""
    conf: float = 0.0
    extra: Optional[Dict[str, Any]] = None

    def xyxy(self, img_w: int, img_h: int) -> Tuple[float, float, float, float]:
        bw, bh = self.w * img_w, self.h * img_h
        cx, cy = self.xc * img_w, self.yc * img_h
        return (cx - bw / 2.0, cy - bh / 2.0, cx + bw / 2.0, cy + bh / 2.0)

    def clipped(self) -> "Box":
        x = max(0.0, min(1.0, self.xc))
        y = max(0.0, min(1.0, self.yc))
        w = max(0.0, min(1.0, self.w))
        h = max(0.0, min(1.0, self.h))
        return Box(self.cls_id, x, y, w, h, self.source, self.conf, self.extra or {})


def xyxy_to_norm(x1: float, y1: float, x2: float, y2: float, img_w: int, img_h: int) -> Tuple[float, float, float, float]:
    x1 = max(0.0, min(float(img_w - 1), float(x1)))
    y1 = max(0.0, min(float(img_h - 1), float(y1)))
    x2 = max(0.0, min(float(img_w - 1), float(x2)))
    y2 = max(0.0, min(float(img_h - 1), float(y2)))
    if x2 < x1:
        x1, x2 = x2, x1
    if y2 < y1:
        y1, y2 = y2, y1
    bw, bh = max(0.0, x2 - x1), max(0.0, y2 - y1)
    return ((x1 + x2) / 2.0 / max(1, img_w), (y1 + y2) / 2.0 / max(1, img_h), bw / max(1, img_w), bh / max(1, img_h))


def norm_to_xyxy(xc: float, yc: float, w: float, h: float, img_w: int, img_h: int) -> Tuple[float, float, float, float]:
    bw, bh = w * img_w, h * img_h
    cx, cy = xc * img_w, yc * img_h
    return (cx - bw / 2.0, cy - bh / 2.0, cx + bw / 2.0, cy + bh / 2.0)


def iou_xyxy(a: Sequence[float], b: Sequence[float]) -> float:
    ax1, ay1, ax2, ay2 = [float(x) for x in a]
    bx1, by1, bx2, by2 = [float(x) for x in b]
    ix1, iy1 = max(ax1, bx1), max(ay1, by1)
    ix2, iy2 = min(ax2, bx2), min(ay2, by2)
    iw, ih = max(0.0, ix2 - ix1), max(0.0, iy2 - iy1)
    inter = iw * ih
    area_a = max(0.0, ax2 - ax1) * max(0.0, ay2 - ay1)
    area_b = max(0.0, bx2 - bx1) * max(0.0, by2 - by1)
    denom = area_a + area_b - inter
    return inter / denom if denom > 0 else 0.0


@dataclass
class Match:
    pred_idx: int
    ref_idx: int
    iou: float


def greedy_match(preds: List[Box], refs: List[Box], img_w: int, img_h: int, iou_thr: float = 0.3) -> List[Match]:
    pairs: List[Tuple[float, int, int]] = []
    for pi, p in enumerate(preds):
        for ri, r in enumerate(refs):
            if int(p.cls_id) != int(r.cls_id):
                continue
            v = iou_xyxy(p.xyxy(img_w, img_h), r.xyxy(img_w, img_h))
            if v >= iou_thr:
                pairs.append((v, pi, ri))
    pairs.sort(reverse=True)
    used_p, used_r, out = set(), set(), []
    for v, pi, ri in pairs:
        if pi in used_p or ri in used_r:
            continue
        used_p.add(pi)
        used_r.add(ri)
        out.append(Match(pi, ri, v))
    return out

    if path is None or not path.exists():
        return []
    boxes: List[Box] = []
    for line in path.read_text(encoding="utf-8", errors="ignore").splitlines():
        line = line.strip()
        if not line or line.startswith("#"):
            continue
        parts = line.split()
        if len(parts) < 5:
            continue
        cls_id = safe_int(parts[0], -1)

        xc, yc, w, h = [safe_float(x, 0.0) for x in parts[1:5]]
        conf = safe_float(parts[5], 0.0) if len(parts) > 5 else 0.0
        boxes.append(Box(cls_id, xc, yc, w, h, source=source, conf=conf, extra={}))
    return boxes

def crop_box_with_context(img: Image.Image, x1: float, y1: float, x2: float, y2: float, expand_ratio: float) -> Tuple[Image.Image, List[int]]:
    w, h = img.size
    bw = max(1.0, x2 - x1)
    bh = max(1.0, y2 - y1)
    cx = (x1 + x2) / 2.0
    cy = (y1 + y2) / 2.0
    nw = bw * expand_ratio
    nh = bh * expand_ratio
    nx1 = clamp(cx - nw / 2.0, 0.0, float(max(0, w - 1)))
    ny1 = clamp(cy - nh / 2.0, 0.0, float(max(0, h - 1)))

    ny2 = clamp(cy + nh / 2.0, 0.0, float(max(1, h - 1)))
    ix1, iy1, ix2, iy2 = int(nx1), int(ny1), int(max(nx1 + 1, nx2)), int(max(ny1 + 1, ny2))

RE_SCORE_JSON = re.compile(r'"score"\s*:\s*([0-9]*\.?[0-9]+)')
RE_JSON_BLOCK = re.compile(r"\{.*\}", flags=re.S)
REJECTED_PATTERNS = [re.compile(p) for p in [r"\brejected\b", r"\breject\b", r"\bnegative\b", r"\bfalse\b", r"误检", r"排除", r"不存在", r"没有", r"不是", r"not\s+present", r"no\s+fire", r"no\s+smoke"]]
CONFIRMED_PATTERNS = [re.compile(p) for p in [r"\bconfirmed\b", r"\bconfirm\b", r"\bpositive\b", r"\byes\b", r"确认", r"确实", r"存在明显", r"可以看到", r"有明显", r"有火", r"有烟"]]
SUSPICIOUS_PATTERNS = [re.compile(p) for p in [r"\bsuspicious\b", r"\buncertain\b", r"\bmaybe\b", r"\bunclear\b", r"疑似", r"不确定", r"可能", r"看不清"]]


def try_extract_json(text: str) -> Optional[Dict[str, Any]]:
    text = (text or "").strip()
    if not text:
        return None
    try:
        obj = json.loads(text)
        if isinstance(obj, dict):
            return obj
    except Exception:
        pass
    m = RE_JSON_BLOCK.search(text)
    if not m:
        return None
    try:
        obj = json.loads(m.group(0))
        return obj if isinstance(obj, dict) else None
    except Exception:
        return None


def normalize_conclusion(value: Any) -> str:
    if value is None:
        return "suspicious"
    text = str(value).strip().lower()
    if text in ["confirmed", "confirm", "positive", "yes", "true", "accept", "accepted", "存在", "确认"]:
        return "confirmed"
    if text in ["rejected", "reject", "negative", "no", "false", "deny", "denied", "误检", "排除", "不存在"]:
        return "rejected"
    return "suspicious"


def fallback_parse(output_text: str) -> Tuple[str, float, str]:
    text = (output_text or "").strip().lower()
    has_rejected = any(p.search(text) for p in REJECTED_PATTERNS)
    has_confirmed = any(p.search(text) for p in CONFIRMED_PATTERNS)
    has_suspicious = any(p.search(text) for p in SUSPICIOUS_PATTERNS)
    m = RE_SCORE_JSON.search(text)
    parsed_score = safe_float(m.group(1), 0.0) if m else None
    if has_confirmed and not has_rejected:
        return "confirmed", max(0.72, min(0.98, parsed_score if parsed_score is not None else 0.82)), "fallback_confirmed"
    if has_rejected and not has_confirmed:
        return "rejected", min(0.35, parsed_score if parsed_score is not None else 0.18), "fallback_rejected"
    if has_suspicious or (has_confirmed and has_rejected):
        return "suspicious", max(0.35, min(0.70, parsed_score if parsed_score is not None else 0.50)), "fallback_suspicious"
    return "suspicious", 0.50, "fallback_default"


def parse_prediction(output_text: str) -> Tuple[str, float, str, bool]:
    obj = try_extract_json(output_text)
    if obj is not None:
        conclusion = normalize_conclusion(obj.get("conclusion", "suspicious"))
        score = max(0.0, min(1.0, safe_float(obj.get("score", 0.5), 0.5)))
        reason = str(obj.get("reason", ""))[:80]
        return conclusion, score, reason, True
    conclusion, score, reason = fallback_parse(output_text)
    return conclusion, score, reason, False

    if out["y2"] < out["y1"]:

def build_recovery_prompt(cls_name: str) -> str:

class VLMFO1Reviewer:
    def __init__(self, model_path: str, code_dir: Optional[str], top_p: float, temperature: float, max_new_tokens: int, recovery_max_new_tokens: int, use_autocast: bool = True, autocast_dtype: str = "bfloat16") -> None:
        self.model_path = str(model_path)
        self.code_dir = str(code_dir) if code_dir else None
        self.top_p = float(top_p)
        self.temperature = float(temperature)
        self.max_new_tokens = int(max_new_tokens)
        self.recovery_max_new_tokens = int(recovery_max_new_tokens)
        self.use_autocast = bool(use_autocast)
        self.autocast_dtype = str(autocast_dtype)
        if self.code_dir and self.code_dir not in sys.path:
            sys.path.insert(0, self.code_dir)
        try:
            from vlm_fo1.model.builder import load_pretrained_model
            from vlm_fo1.mm_utils import prepare_inputs
            from vlm_fo1.task_templates import OD_template
        except Exception as e:
            raise ImportError("无法导入 VLM-FO1。请确认 --code-dir 指向 VLM-FO1 仓库根目录，并且依赖已安装。") from e
        self._load_pretrained_model = load_pretrained_model
        self._prepare_inputs = prepare_inputs
        self._OD_template = OD_template
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.tokenizer, self.model, self.image_processors = self._safe_load_model()
        self.model.eval()

    def _safe_load_model(self):
        try:
            return self._load_pretrained_model(model_path=self.model_path, device=self.device)
        except TypeError:
            return self._load_pretrained_model(self.model_path)

    def _autocast_ctx(self):
        if self.device != "cuda" or not self.use_autocast:
            return torch.autocast(device_type="cpu", enabled=False)
        dtype = torch.bfloat16 if self.autocast_dtype == "bfloat16" else torch.float16
        return torch.autocast(device_type="cuda", dtype=dtype)

    def _decode_outputs(self, output_ids, generation_kwargs) -> str:
        prompt_len = None
        if isinstance(generation_kwargs, dict):
            if "inputs" in generation_kwargs and hasattr(generation_kwargs["inputs"], "shape"):
                prompt_len = int(generation_kwargs["inputs"].shape[1])
            elif "input_ids" in generation_kwargs and hasattr(generation_kwargs["input_ids"], "shape"):
                prompt_len = int(generation_kwargs["input_ids"].shape[1])
        if prompt_len is None:
            return self.tokenizer.decode(output_ids[0], skip_special_tokens=True).strip()
        return self.tokenizer.decode(output_ids[0, prompt_len:], skip_special_tokens=True).strip()

    def _run_one_pass(self, image_path: str, bbox_list: List[List[float]], prompt_text: str, max_tokens: int) -> str:
        messages = [{"role": "user", "content": [{"type": "image_url", "image_url": {"url": image_path}}, {"type": "text", "text": self._OD_template.format(prompt_text)}], "bbox_list": bbox_list}]
        kwargs = {"max_tokens": max_tokens, "do_sample": False}
        try:
            generation_kwargs = self._prepare_inputs(self.model_path, self.model, self.image_processors, self.tokenizer, messages, **kwargs)
        except TypeError:
            generation_kwargs = self._prepare_inputs(self.model_path, self.model, self.image_processors, self.tokenizer, messages, max_tokens=max_tokens, top_p=self.top_p, temperature=self.temperature, do_sample=False)
        with torch.inference_mode():
            with self._autocast_ctx():
                output_ids = self.model.generate(**generation_kwargs)
        return self._decode_outputs(output_ids, generation_kwargs)

    def review_box(self, image_path: str, box_xyxy: List[int], cls_name: str, review_stage: str, crop_path: Optional[str] = None, crop_box_xyxy: Optional[List[int]] = None, enable_recovery: bool = True) -> Dict[str, Any]:
        raw_main = self._run_one_pass(image_path, [list(map(float, box_xyxy))], build_main_prompt(cls_name, review_stage), self.max_new_tokens)
        conclusion, score, reason, parsed_as_json = parse_prediction(raw_main)
        recovery_used = 0
        raw_recovery = ""
        rec_conclusion = rec_score = rec_reason = rec_parsed = None
        if enable_recovery and crop_path and crop_box_xyxy and collapse_like(conclusion, score, raw_main, parsed_as_json):
            recovery_used = 1
            raw_recovery = self._run_one_pass(crop_path, [list(map(float, crop_box_xyxy))], build_recovery_prompt(cls_name), self.recovery_max_new_tokens)
            rec_conclusion, rec_score, rec_reason, rec_parsed = parse_prediction(raw_recovery)
            if normalize_conclusion(rec_conclusion) in {"confirmed", "rejected"}:
                conclusion, score, reason, parsed_as_json = rec_conclusion, rec_score, rec_reason, rec_parsed
            elif normalize_conclusion(conclusion) == "suspicious" and normalize_conclusion(rec_conclusion) == "suspicious":
                if abs((rec_score or 0.5) - 0.5) > abs(score - 0.5):
                    conclusion, score, reason, parsed_as_json = rec_conclusion, rec_score, rec_reason, rec_parsed
        return {"conclusion": normalize_conclusion(conclusion), "score": float(score), "reason": str(reason), "parsed_as_json": int(bool(parsed_as_json)), "raw_main_output": raw_main, "recovery_used": recovery_used, "raw_recovery_output": raw_recovery, "recovery_conclusion": rec_conclusion, "recovery_score": rec_score, "recovery_reason": rec_reason, "recovery_parsed_as_json": int(bool(rec_parsed)) if rec_parsed is not None else 0}

def decision_from_conclusion(conclusion: str, score: float, risk_score: float, policy: str, reject_score_thr: float, default_uncertain: str) -> Tuple[str, int, int]:
    """
    增强版决策：引入上游 box_risk_score 进行交叉否决。
    规则：VLM 模糊(suspicious) + 上游高风险(≥0.45) → 直接 reject
    """
    c = normalize_conclusion(conclusion)
    thr = float(reject_score_thr)
    
    # ⚠️ 可调参数：上游风险分拦截阈值
    # 如果你的 nofire 仍然漏放，可调低至 0.40；如果误杀偏多，可调高至 0.50
    RISK_THR = 0.45 

    if policy == "strict":
        if c == "confirmed":
            return "accept", 1, 1
        if c == "rejected":
            return "reject", 0, 0
        return default_uncertain, int(default_uncertain == "accept"), int(default_uncertain == "accept")

    # filter mode: only clear negative is vetoed.
    if c == "rejected" and float(score) <= thr:
        return "reject", 0, 0
    if c == "confirmed":
        return "accept", 1, 1
        
    # 🔑 方案C核心：VLM犹豫 + 上游高风险/低自信 = 强误报信号
    if c == "suspicious":
        if float(score) <= 0.40 or float(risk_score) >= RISK_THR:
            return "reject", 0, 0
            
    return "uncertain", 1, 1


def rules_review(row: Dict[str, Any]) -> Dict[str, Any]:
    risk_score = float(row.get("box_risk_score", 0.0) or 0.0)
    if row.get("review_stage") == "after_sam":
        return {"conclusion": "suspicious", "score": 0.50, "reason": "rules_after_sam_keep", "parsed_as_json": 1, "raw_main_output": "", "recovery_used": 0, "raw_recovery_output": ""}
    if risk_score >= 0.85:
        return {"conclusion": "rejected", "score": 0.20, "reason": "rules_high_risk", "parsed_as_json": 1, "raw_main_output": "", "recovery_used": 0, "raw_recovery_output": ""}
    return {"conclusion": "suspicious", "score": 0.50, "reason": "rules_keep", "parsed_as_json": 1, "raw_main_output": "", "recovery_used": 0, "raw_recovery_output": ""}

    errors: List[Dict[str, Any]] = []

    # 2. 使用 tqdm 包装 rows，并结合 enumerate
    # desc: 进度条左侧提示文字
    # unit: 进度单位
    pbar = tqdm(rows, desc="Reviewing Boxes", unit="box")
    
    for idx, row in enumerate(pbar, 1):
        # 3. 动态更新进度条右侧的状态 (显示当前的 sample_key，方便定位)
        current_key = str(row.get("sample_key", "unknown"))[:10]
        pbar.set_postfix({"sample": current_key})

        crop_path = ""
        crop_box_xyxy: Optional[List[int]] = None
        
        try:
            x1, y1, x2, y2 = [safe_float(row.get(k), 0.0) for k in ["x1", "y1", "x2", "y2"]]
            box_xyxy = [int(round(x1)), int(round(y1)), int(round(x2)), int(round(y2))]

                    str(row.get("review_stage", "")), crop_path or None,

                )

            # 🚀 [方案C] 注入上游风险分
            risk_score = float(row.get("box_risk_score", 0.0))

                result["conclusion"],
                float(result["score"]),
                risk_score,
                args.decision_policy,
                args.reject_score_thr,

            verdict = {

            }

            review_pack.append({**row, "crop_path": crop_path, "crop_box_xyxy": json.dumps(crop_box_xyxy or [], ensure_ascii=False)})

        except Exception as e:
            # 4. 遇到错误时，使用 logger.error 记录，同时避免进度条错乱
            # tqdm.write 可以让日志输出不破坏进度条的样式
            error_msg = f"Error at index {idx}, UID {row.get('box_uid')}: {str(e)}"
            tqdm.write(f"[ERROR] {error_msg}") 
            
            err = {"row_idx": idx, "box_uid": row.get("box_uid", ""), "error": str(e), "traceback": traceback.format_exc()}
            errors.append(err)
            
            if not args.continue_on_error:
                logger.critical("Stopping due to error.")
                raise

            # 异常处理分支同步注入风险分
            risk_score = float(row.get("box_risk_score", 0.0))
            decision, keep_train, add_final = decision_from_conclusion(
                "suspicious", 0.50, risk_score, args.decision_policy, args.reject_score_thr, args.default_uncertain_decision
            )
            verdicts.append({
                **row, "vlm_decision": decision, "vlm_conclusion": "suspicious", 
                "vlm_score": 0.50, "vlm_reason": "error_keep_by_filter" if args.decision_policy == "filter" else "error", 
                "vlm_model": args.mode, "decision_policy": args.decision_policy, 
                "should_keep_for_training": keep_train, "should_add_to_final_labels": add_final
            })

    # 5. 保存结果
    logger.info("Saving results...")
    write_csv(out_dir / "vlm_box_verdicts.csv", verdicts)
    write_csv(out_dir / "review_pack.csv", review_pack)
    write_csv(out_dir / "errors.csv", errors)
    
    diagnostics = {
        "input_candidates": len(rows), 
        "verdicts": len(verdicts), 
        "errors": len(errors), 
        "decision_counts": dict(Counter([v.get("vlm_decision", "") for v in verdicts])), 
        "keep_final_counts": dict(Counter([str(v.get("should_add_to_final_labels", "")) for v in verdicts]))
    }
    write_json(out_dir / "diagnostics.json", diagnostics)
    write_json(out_dir / "meta.json", {"mode": args.mode, "decision_policy": args.decision_policy, "candidate_csvs": args.candidate_csvs, "output_dir": str(out_dir)})

    # 6. 最终总结输出
    logger.info("=" * 80)
    logger.info("[DONE] VLM box review finished")
    logger.info(json.dumps(diagnostics, ensure_ascii=False, indent=2))


if __name__ == "__main__":

    base_root = Path(args.base_label_root)
    out_dir = Path(args.output_dir)
    out_label_root = out_dir / "labels"
    report_dir = out_dir / "report"

    ensure_dir(out_label_root)
    ensure_dir(report_dir)

    verdict_by_uid, keep_uids, veto_uids = load_verdicts(Path(args.verdict_csv))
    after_sam_rows = read_candidate_rows(args.after_sam_candidates)
    after_sam_by_sample = group_rows_by_sample(after_sam_rows)

    label_files = sorted(base_root.rglob("*.txt"))

    written_rows: List[Dict[str, Any]] = []
    summary_counter = Counter()
    existing_samples: Set[str] = set()

    # Pass 1: process every YOLO txt
    for label_path in label_files:
        sample_key = make_sample_key_from_label_path(label_path, base_root)
        existing_samples.add(sample_key)

        base_boxes = load_yolo_labels(label_path, source="yolo")
        filtered_yolo_boxes = filter_yolo_boxes_by_vlm(sample_key, base_boxes, veto_uids)

        summary_counter["yolo_boxes_input"] += len(base_boxes)
        summary_counter["yolo_boxes_vlm_rejected"] += max(len(base_boxes) - len(filtered_yolo_boxes), 0)

        if filtered_yolo_boxes:
            # If at least one YOLO box remains after VLM filtering, YOLO wins.
            final_boxes = filtered_yolo_boxes
            strategy = "vlm_filtered_yolo_only"
            summary_counter["samples_yolo_wins_after_vlm"] += 1
            summary_counter["yolo_boxes_kept_after_vlm"] += len(filtered_yolo_boxes)
        else:
            # Only when no YOLO box remains after VLM filtering do we fallback to kept SAM boxes.
            final_boxes = kept_sam_boxes_for_sample(sample_key, after_sam_by_sample, keep_uids)

            if final_boxes:
                strategy = "sam_fallback_after_yolo_empty_or_all_rejected"
                summary_counter["samples_sam_fallback"] += 1
                summary_counter["sam_boxes_used"] += len(final_boxes)
            else:
                strategy = "empty"
                summary_counter["samples_empty"] += 1

        out_path = out_label_root / label_path.relative_to(base_root)
        write_yolo_labels(out_path, final_boxes, with_conf=False)

        written_rows.append(
            {
                "sample_key": sample_key,
                "input_label_path": str(label_path),
                "output_label_path": str(out_path),
                "base_count": len(base_boxes),
                "yolo_after_vlm_count": len(filtered_yolo_boxes),
                "final_count": len(final_boxes),
                "merge_strategy": strategy,
            }
        )

    # Pass 2: create labels for samples missing from YOLO txts entirely
    if args.allow_new_sam_samples:
        for sample_key, rows in after_sam_by_sample.items():
            if sample_key in existing_samples:
                continue

            kept_rows: List[Dict[str, Any]] = []
            for r in rows:
                uid = str(r.get("box_uid", "")).strip()
                if uid and uid in keep_uids:
                    kept_rows.append(r)

            if not kept_rows:
                continue

            final_boxes = [candidate_to_box(r) for r in kept_rows]
            out_path = out_label_root / f"{sample_key}.txt"
            write_yolo_labels(out_path, final_boxes, with_conf=False)

            written_rows.append(
                {
                    "sample_key": sample_key,
                    "input_label_path": "",
                    "output_label_path": str(out_path),
                    "base_count": 0,
                    "yolo_after_vlm_count": 0,
                    "final_count": len(final_boxes),
                    "merge_strategy": "sam_new_sample_no_yolo_file",
                }
            )

            summary_counter["samples_sam_new_file"] += 1
            summary_counter["sam_boxes_used_new_file"] += len(final_boxes)

    summary = {
        "labels_written": len(written_rows),
        "counts": dict(summary_counter),
        "verdicts_total": len(verdict_by_uid),
        "verdict_keep": len(keep_uids),
        "verdict_veto": len(veto_uids),
        "output_label_root": str(out_label_root),
        "merge_policy": {
            "step_1": "filter_yolo_by_vlm",
            "if_filtered_yolo_has_boxes": "use_filtered_yolo_only",
            "if_filtered_yolo_has_no_boxes": "use_vlm_kept_sam_only",
            "if_no_yolo_txt_exists": "create_new_txt_only_when_vlm_kept_sam_exists_and_allow_new_sam_samples_is_true",
        },
    }

    write_csv(report_dir / "merge_manifest.csv", written_rows)
    write_json(report_dir / "merge_summary.json", summary)

    print("=" * 80)
    print("[DONE] merge finished")
    print(json.dumps(summary, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()

def save_yaml(output_path: Path, data: dict):
    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w", encoding="utf-8") as f:

def get_env_desc() -> str:
    if torch.cuda.is_available() and torch.cuda.device_count() > 0:

            gpu_name = "Unknown GPU"
    else:
        gpu_name = "CPU"
    return f"PyTorch {torch.__version__} | CUDA {torch.cuda.is_available()} | Device {gpu_name}"


def apply_optional_override(target: dict, key: str, value):
    if value is not None:
        target[key] = value

    """

        "flipud", "fliplr", "mosaic", "mixup", "copy_paste",

        apply_optional_override(hyp, key, getattr(args, key))

    return hyp
